-- catalogue CORA : Analyse statistique - listes de travail MCO / bébés nés par cesarienne
SELECT      SB.NO_SEJOUR AS "IEP BEBE" , 
    PB.NOM_USUEL_PATIENT AS "NOM BEBE" , 
    PB.PRENOM_PATIENT AS "PRENOM BEBE" , 
    PB.DATE_NAIS AS "DATE NAIS BEBE" , 
    SM.NO_SEJOUR AS "IEP MAMAN" , 
    PM.NOM_USUEL_PATIENT AS "NOM MAMAN" , 
    PM.PRENOM_PATIENT AS "PRENOM MAMAN" , 
    PM.DATE_NAIS AS "DATE NAIS MAMAN" , 
    A.DATE_ACTE AS "DATE ACCOUCHEMENT" , 
    A.CODE_ACTE AS "CODE ACTE" , 
    EG.CODE_EG AS "CODE EG" , 
    EG.LIBELLE_EG AS "LIBELLE EG" 

FROM     TB_PATIENT PB
     INNER JOIN TB_SEJOUR SB
         INNER JOIN TB_SYNTH_RSS RSSB
          ON  SB.ID_SEJOUR = RSSB.ID_SEJOUR
      ON  SB.ID_PATIENT = PB.ID_PATIENT
     INNER JOIN TB_SEJOUR SM
         INNER JOIN TB_MVT M
             INNER JOIN TB_ACTE A
              ON  M.ID_MVT = A.ID_MVT
          ON  SM.ID_SEJOUR = M.ID_SEJOUR
         LEFT JOIN TB_EG_UNITE EGU
             INNER JOIN TB_EG EG
              ON  EGU.ID_EG = EG.ID_EG
          ON  M.ID_UNITE_PMSI = EGU.ID_UNITE
      ON  SM.NO_SEJOUR = SB.NO_SEJOUR_DOSSIER_MERE
     INNER JOIN TB_PATIENT PM
      ON  PM.ID_PATIENT = SM.ID_PATIENT

WHERE 
1=1 
  AND A.CODE_ACTE IN ('JQGA002','JQGA003','JQGA004','JQGA005') AND SM.DATE_SORTIE >= '01/07/2026 00:00' AND SM.DATE_SORTIE <= '31/07/2026 23:59'
