-- catalogue CORA : Analyse statistique - listes de travail MCO / les statistiques / detail des actes (interventions)
SELECT      A.DATE_ACTE AS "DATE ACTE" , 
    RTRIM(P.NOM_USUEL_PATIENT) || ' ' || RTRIM(P.PRENOM_PATIENT) AS "PATIENT" , 
    P.DATE_NAIS AS "DATE NAIS" , 
    S.NO_SEJOUR AS "NO SEJOUR" , 
    A.HEURE_ACTE AS "HEURE ACTE" , 
    A.CODE_ACTE AS "CODE ACTE" , 
    CCAM.LIBELLE_LONG AS "LIBELLE ACTE" , 
    RTRIM(UEXEC.NOM_UTIL) || ' ' || RTRIM(UEXEC.PRENOM_UTIL) AS "EXECUTANT" , 
    RTRIM(USIGN.CODE_UTIL) || ' (' || to_char(DA.HEURE_SIGN,'DD/MM/YYYY HH24:MI:SS') || ')' AS "SIGNATAIRE" , 
    CASE WHEN DA.CODE_ASSOCIATION = '2' THEN '50%' WHEN DA.CODE_ASSOCIATION = '3' THEN '75%' WHEN DA.FACTURABLE = 'N' THEN '0%' ELSE '100%' END AS "ANP" , 
    DA.MODIFICATEUR AS "MODIFICATEURS" , 
    DA.MNT_ACTIVITE AS "MONTANT ACTIVITE (€)" , 
    DA.MNT_DEPASS AS "DEPASSEMENT (€)" , 
    DA.MNT_TOTAL AS "TOTAL (€)" , 
    DA.ACTIVITE_LIBERALE AS "ACT LIBERALE" , 
    A.NO_INTERVENTION AS "NO INTERVENTION" 

FROM     TB_PATIENT P
     INNER JOIN TB_SEJOUR S
         INNER JOIN TB_MVT M
             INNER JOIN TB_ACTE A
                 INNER JOIN TB_DET_ACTE DA
                     LEFT JOIN CORA_UTIL.UTIL USIGN
                      ON  DA.ID_MEDECIN_SIGN = USIGN.ID_UTIL
                     LEFT JOIN CORA_UTIL.UTIL UEXEC
                      ON  DA.ID_MEDECIN = UEXEC.ID_UTIL
                  ON  A.ID_ACTE = DA.ID_ACTE
                 INNER JOIN CORA_CCAM.CCAM_NV CCAM
                  ON  A.CODE_ACTE = CCAM.CODE
              ON  M.ID_MVT = A.ID_MVT
          ON  S.ID_SEJOUR = M.ID_SEJOUR
      ON  P.ID_PATIENT = S.ID_PATIENT

WHERE 
1=1 
  AND A.DATE_ACTE >= '01/07/2026 00:00' AND A.DATE_ACTE <= '31/07/2026 23:59' AND UEXEC.CODE_UTIL = :CODE_UTIL AND A.NO_INTERVENTION > 0
 ORDER BY A.DATE_ACTE ASC, RTRIM(P.NOM_USUEL_PATIENT) || ' ' || RTRIM(P.PRENOM_PATIENT) ASC, A.HEURE_ACTE ASC, A.CODE_ACTE ASC
