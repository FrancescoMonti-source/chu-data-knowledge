-- catalogue CORA : Analyse statistique - listes de travail MCO / les statistiques / palmares des actes
SELECT      A.CODE_ACTE AS "CODE ACTE" , 
    CCAM.LIBELLE_LONG AS "LIBELLE ACTE" , 
    A.DATE_ACTE AS "DATE ACTE" , 
    S.CODE_SITE AS "CODE SITE" , 
    S.LIBELLE_SITE AS "LIBELLE SITE" , 
    A.ID_ACTE AS "NB ACTES" 

FROM     TB_ACTE A
     INNER JOIN TB_DET_ACTE DA
         LEFT JOIN CORA_UTIL.UTIL UEXEC
          ON  DA.ID_MEDECIN = UEXEC.ID_UTIL
      ON  A.ID_ACTE = DA.ID_ACTE
     INNER JOIN CORA_CCAM.CCAM_NV CCAM
      ON  A.CODE_ACTE = CCAM.CODE
     INNER JOIN TB_SITE S
      ON  A.ID_SITE = S.ID_SITE

WHERE 
1=1 
  AND A.DATE_ACTE >= '01/07/2026 00:00' AND A.DATE_ACTE <= '31/07/2026 23:59' AND UEXEC.CODE_UTIL = :CODE_UTIL
