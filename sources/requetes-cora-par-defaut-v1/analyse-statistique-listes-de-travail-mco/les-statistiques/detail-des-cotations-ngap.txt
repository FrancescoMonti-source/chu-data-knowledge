-- catalogue CORA : Analyse statistique - listes de travail MCO / les statistiques / detail des cotations NGAP
SELECT      NGAP.DATE_ACTE AS "DATE ACTE" , 
    RTRIM(P.NOM_USUEL_PATIENT) || ' ' || RTRIM(P.PRENOM_PATIENT) AS "PATIENT" , 
    P.DATE_NAIS AS "DATE NAIS" , 
    S.NO_SEJOUR AS "NO SEJOUR" , 
    NGAP.DATE_ACTE AS "HEURE ACTE" , 
    LC.LETTRE_CLE AS "LETTRE CLE" , 
    LC.COEF AS "COEF" , 
    NGAP.LIBELLE_COTATION AS "LIBELLE ACTE" , 
    RTRIM(UEXEC.NOM_UTIL) || ' ' || RTRIM(UEXEC.PRENOM_UTIL) AS "EXECUTANT" , 
    NVL(LC.MNT_LC,0) AS "MONTANT LC (€)" , 
    NVL(LC.MNT_DEPASS_LC,0) AS "DEPASSEMENT (€)" , 
    NVL(LC.MNT_LC,0) + NVL(LC.MNT_DEPASS_LC,0) AS "TOTAL (€)" , 
    ROUND(NVL(TR.TAUX_REDEVANCE,0) * (NVL(LC.MNT_LC,0) + NVL(LC.MNT_DEPASS_LC,0)),2) AS "REDEVANCE (€)" , 
    NGAP.ACT_PRIVE AS "ACT LIBERALE" 

FROM     TB_PATIENT P
     INNER JOIN TB_SEJOUR S
         INNER JOIN TB_MVT M
             INNER JOIN TB_COTATION_NGAP NGAP
                 INNER JOIN TB_NGAP_LC LC
                     LEFT JOIN TB_TAUX_REDEVANCE TR
                      ON  LC.LETTRE_CLE = TR.CODE_TAUX_REDEVANCE
                  ON  NGAP.CPT_ID_COTATION_NGAP = LC.CPT_ID_COTATION_NGAP
                 LEFT JOIN CORA_UTIL.UTIL UEXEC
                  ON  NGAP.ID_MEDECIN = UEXEC.ID_UTIL
              ON  M.ID_MVT = NGAP.ID_MVT
          ON  S.ID_SEJOUR = M.ID_SEJOUR
      ON  P.ID_PATIENT = S.ID_PATIENT

WHERE 
1=1 
  AND TRUNC(NGAP.DATE_ACTE) >= '01/07/2026 00:00' AND TRUNC(NGAP.DATE_ACTE) <= '31/07/2026 23:59' AND UEXEC.CODE_UTIL = :CODE_UTIL AND (TR.DATE_DEBUT <= NGAP.DATE_ACTE OR TR.DATE_DEBUT IS NULL) AND (TR.DATE_FIN >= NGAP.DATE_ACTE OR TR.DATE_FIN IS NULL)
 ORDER BY NGAP.DATE_ACTE ASC, RTRIM(P.NOM_USUEL_PATIENT) || ' ' || RTRIM(P.PRENOM_PATIENT) ASC, NGAP.COTATION ASC
