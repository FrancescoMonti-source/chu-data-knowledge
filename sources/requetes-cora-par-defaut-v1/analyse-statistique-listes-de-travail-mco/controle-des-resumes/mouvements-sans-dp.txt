-- catalogue CORA : Analyse statistique - listes de travail MCO / controle des résumés / mouvements sans DP
SELECT   DISTINCT     P.NOM_USUEL_PATIENT AS "NOM USUEL PATIENT" , 
    P.PRENOM_PATIENT AS "PRENOM PATIENT" , 
    P.DATE_NAIS AS "DATE NAIS" , 
    S.NO_SEJOUR AS "NUM SEJOUR" , 
    M.HEURE_DEBUT_MVT AS "DATE DEBUT MVT" , 
    M.HEURE_FIN_MVT AS "DATE FIN MVT" , 
    U.CODE_UNITE AS "UF PMSI" , 
    vueMvtCompl.CODE_UNITE AS "UF COMPL" , 
    U.LIBELLE_UNITE AS "LIB UF PMSI" , 
    RTRIM(UTIL.NOM_UTIL) || ' ' || RTRIM(UTIL.PRENOM_UTIL) AS "MED RESP MVT" 

FROM     TB_PATIENT P
     INNER JOIN TB_SEJOUR S
         INNER JOIN TB_MVT M
             INNER JOIN TB_UNITE U
              ON  M.ID_UNITE_PMSI = U.ID_UNITE
             LEFT JOIN (SELECT MC.ID_SEJOUR, MC.ID_UNITE_PMSI, MC.CODE_UNITE    FROM    TB_MVT_COMPL MC INNER    JOIN                TB_MVT M ON M.ID_SEJOUR = MC.ID_SEJOUR    WHERE    (((to_date(to_char(M.DATE_DEBUT_MVT,'DD/MM/YYYY'),'DD/MM/YYYY')=to_date(to_char(MC.DATE_DEBUT,'DD/MM/YYYY'),'DD/MM/YYYY') AND to_date(to_char(M.HEURE_DEBUT_MVT,'HH24:MI'),'HH24:MI')>=to_date(to_char(MC.HEURE_DEBUT,'HH24:MI'),'HH24:MI') OR to_date(to_char(M.DATE_DEBUT_MVT,'DD/MM/YYYY'),'DD/MM/YYYY')>to_date(to_char(MC.DATE_DEBUT,'DD/MM/YYYY'),'DD/MM/YYYY')) AND (to_date(to_char(M.DATE_DEBUT_MVT,'DD/MM/YYYY'),'DD/MM/YYYY')=to_date(to_char(MC.DATE_FIN,'DD/MM/YYYY'),'DD/MM/YYYY') AND to_date(to_char(M.HEURE_DEBUT_MVT,'HH24:MI'),'HH24:MI')<to_date(to_char(MC.HEURE_FIN,'HH24:MI'),'HH24:MI') OR to_date(to_char(M.DATE_DEBUT_MVT,'DD/MM/YYYY'),'DD/MM/YYYY')<to_date(to_char(MC.DATE_FIN,'DD/MM/YYYY'),'DD/MM/YYYY') OR MC.DATE_FIN IS NULL)) OR (M.DATE_FIN_MVT IS NULL AND MC.DATE_FIN IS NULL OR ((to_date(to_char(M.DATE_FIN_MVT,'DD/MM/YYYY'),'DD/MM/YYYY')=to_date(to_char(MC.DATE_DEBUT,'DD/MM/YYYY'),'DD/MM/YYYY') AND to_date(to_char(M.HEURE_FIN_MVT,'HH24:MI'),'HH24:MI')>to_date(to_char(MC.HEURE_DEBUT,'HH24:MI'),'HH24:MI') OR to_date(to_char(M.DATE_FIN_MVT,'DD/MM/YYYY'),'DD/MM/YYYY')>to_date(to_char(MC.DATE_DEBUT,'DD/MM/YYYY'),'DD/MM/YYYY')) AND (to_date(to_char(M.DATE_FIN_MVT,'DD/MM/YYYY'),'DD/MM/YYYY')=to_date(to_char(MC.DATE_FIN,'DD/MM/YYYY'),'DD/MM/YYYY') AND to_date(to_char(M.HEURE_FIN_MVT,'HH24:MI'),'HH24:MI')<=to_date(to_char(MC.HEURE_FIN,'HH24:MI'),'HH24:MI') OR to_date(to_char(M.DATE_FIN_MVT,'DD/MM/YYYY'),'DD/MM/YYYY')<to_date(to_char(MC.DATE_FIN,'DD/MM/YYYY'),'DD/MM/YYYY')))))) vueMvtCompl
              ON  M.ID_SEJOUR = vueMvtCompl.ID_SEJOUR
             LEFT JOIN CORA_UTIL.Util UTIL
              ON  M.ID_MEDECIN_RESPONSABLE = UTIL.ID_UTIL
          ON  S.ID_SEJOUR = M.ID_SEJOUR
      ON  P.ID_PATIENT = S.ID_PATIENT

WHERE 
1=1 
  AND NOT EXISTS (SELECT 1       FROM       TB_DIAG D       WHERE       D.ID_MVT IN (SELECT MR1.ID_MVT       FROM       TB_MVT_RUM MR1 INNER JOIN TB_MVT_RUM MR2 ON MR1.ID_RUM = MR2.ID_RUM       WHERE       MR2.ID_MVT = M.ID_MVT) AND D.TYPE_DIAG = 'P') AND NVL(S.STATUT_PROG, 'A') != 'P' AND M.DATE_FIN_MVT >= '01/07/2026 00:00' AND M.DATE_FIN_MVT <= '31/07/2026 23:59' AND NVL(BLOCAGE_RSS, 'N') !='O' AND U.GESTION_PMSI = 'O' AND UTIL.CODE_UTIL = :CODE_UTIL
