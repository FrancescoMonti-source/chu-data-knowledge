-- catalogue CORA : Analyse statistique - listes de travail MCO / interventions bloc non codées
SELECT      P.IPP_PATIENT AS "NUM IPP PATIENT" , 
    P.NOM_USUEL_PATIENT AS "NOM USUEL" , 
    P.NOM_NAISSANCE AS "NOM NAISSANCE" , 
    P.PRENOM_PATIENT AS "PRENOM" , 
    P.SEXE AS "SEXE" , 
    P.DATE_NAIS AS "DATE NAISSANCE" , 
    S.NO_SEJOUR AS "NUM SEJOUR" , 
    to_char(I.DATE_INTERVENTION,'DD/MM/YYYY') AS "DATE INTERVENTION COURTE" , 
    I.NO_INTERVENTION AS "NUM INTERVENTION" , 
    I.ID_INTERVENTION AS "ID INTERVENTION" 

FROM     TB_PATIENT P
     INNER JOIN TB_SEJOUR S
         INNER JOIN TB_MVT M
             INNER JOIN TB_INTERVENTION I
                 LEFT JOIN TB_ACTE A
                  ON  I.ID_INTERVENTION=A.NO_INTERVENTION
                 LEFT JOIN (SELECT OP.ID_INTERVENTION, U.CODE_UTIL AS CODE     FROM      TB_INTERVENTION_OPERATEUR OP, UTIL U       WHERE      U.id_util=OP.ID_OPERATEUR AND OP.TYPE_INTERV='4') VueOp4
                  ON  I.ID_INTERVENTION=VueOp4.ID_INTERVENTION
                 LEFT JOIN (SELECT OP.ID_INTERVENTION, U.CODE_UTIL AS CODE     FROM      TB_INTERVENTION_OPERATEUR OP, UTIL U       WHERE      U.id_util=OP.ID_OPERATEUR AND OP.TYPE_INTERV='1') VueOp1
                  ON  I.ID_INTERVENTION=VueOp1.ID_INTERVENTION
              ON  M.ID_MVT=I.ID_MVT
          ON  S.ID_SEJOUR=M.ID_SEJOUR
      ON  P.ID_PATIENT=S.ID_PATIENT

WHERE 
1=1 
  AND to_char(I.DATE_INTERVENTION,'DD/MM/YYYY') >= '01/01/2026 00:00' AND to_char(I.DATE_INTERVENTION,'DD/MM/YYYY') <= '31/12/2026 23:59' AND (VueOp1.CODE = :CODE_UTIL OR VueOp4.CODE = :CODE_UTIL) AND A.CODE_ACTE IS NULL
