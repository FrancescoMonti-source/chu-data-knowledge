-- catalogue CORA : Certification des comptes / revue de dossier par pole - um (avant dernier)
SELECT      RSSP.CODE_GHM AS "GHM N0" , 
    RSSP.DIAG_P AS "DP N0" , 
    RSSP.CODE_GHS AS "GHS N0" , 
    RSSP.MNT_TOTAL AS "MNT TOT N0" , 
    UTILP.code_util AS "UTIL N0" , 
    RSSD.CODE_GHM AS "GHM Final" , 
    RSSD.DIAG_P AS "DP Final" , 
    RSSD.CODE_GHS AS "GHS Final" , 
    RSSD.MNT_TOTAL AS "MNT TOT Final" , 
    CASE WHEN SUBSTR(RSSP.CODE_GHM,1,5)<>  SUBSTR(RSSD.CODE_GHM,1,5)THEN      CASE WHEN SUBSTR(RSSP.CODE_GHM,6,1)<> SUBSTR(RSSD.CODE_GHM,6,1)THEN 'T' ELSE 'G' END     ELSE CASE WHEN SUBSTR(RSSP.CODE_GHM,6,1)<> SUBSTR(RSSD.CODE_GHM,6,1)THEN 'S' ELSE 'N'  END END AS "MODIF" , 
    UTILD.code_util AS "UTIL Final" , 
    RSS.NO_SEJOUR AS "NO SEJOUR" , 
    RSS.DATE_DEBUT AS "DEBUT" , 
    RSS.DATE_FIN AS "FIN" , 
    RSS.ID_RSS AS "ID_RSS" , 
    UMRSS.CODE_UNITE_MED AS "UM" , 
    UMRSS.LIBELLE_UNITE_MED AS "LIBELLE UM" , 
    PRUM.CODE_POLE AS "POLE" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN TB_SYNTH_RUM RUM
         LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUM.ID_POLE, PRUM.CODE_POLE, PRUM.LIBELLE_POLE, PRUM.ACTIF          FROM          TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUM          WHERE          PUM2.ID_POLE=PRUM.ID_POLE) PRUM
          ON  RUM.ID_UNITE_MEDICALE=PRUM.ID_UNITE_MED
      ON  RSS.ID_RSS=RUM.ID_RSS
     INNER JOIN TB_UNITE_MEDICALE UMRSS
      ON  RSS.ID_UM_RETENU=UMRSS.ID_UNITE_MED
     INNER JOIN TB_RSS_SUIVI RSSD
         INNER JOIN UTIL UTILD
          ON  UTILD.ID_UTIL = RSSD.ID_INFO
      ON  RSSD.ID_RSS=RSS.ID_RSS AND RSSD.ETAT_GROUPAGE='D'
     INNER JOIN TB_RSS_SUIVI RSSP
         INNER JOIN UTIL UTILP
          ON  UTILP.ID_UTIL = RSSP.ID_INFO
      ON  RSSP.ID_RSS=RSS.ID_RSS AND RSSP.CHRONO_GROUPAGE=RSSD.CHRONO_GROUPAGE-1

WHERE 
1=1 
  AND (PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RUM.DATE_FIN_RUM) AND (PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RUM.DATE_FIN_RUM) AND RUM.DP_MVT_RETENU ='X' AND RSS.DATE_FIN>='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
