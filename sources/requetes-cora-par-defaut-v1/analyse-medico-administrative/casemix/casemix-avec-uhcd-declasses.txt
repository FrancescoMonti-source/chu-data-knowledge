SELECT   DISTINCT     RSS.ID_RSS AS "ID_RSS" , 
    RSS.CODE_CMD AS "CODE CMD RSS" , 
    RSS.CODE_CMD || RSS.CODE_GHM AS "GHM RSS" , 
    SUBSTR(RSS.CODE_GHM,-1) AS "SUFFIXE CODE GHM RSS" , 
    RSS.CODE_GHS AS "CODE GHS RSS" , 
    RSS.MNT_TOTAL AS "MONTANT TOTAL CALCULE RSS" , 
    RSS.DUREE_SEJOUR AS "DUREE CORRIGEE RSS" , 
    RSS.MONTANT_GHS AS "MONTANT GHS RSS" , 
    RSS.MNT_TOTAL_SUPPLEMENT AS "MONTANT TOTAL SUPP CALC RSS" , 
    UMRSS.CODE_UNITE_MED AS "CODE UNITE RETENUE RSS" , 
    UMRSS.LIBELLE_UNITE_MED AS "LIBELLE UNITE  RETENUE RSS" , 
    RSS.QUANTITE_J_DEP AS "NB JOURS DEP RSS" , 
    RSS.NB_JOUR_EXB AS "NB JOURS EXB RSS" , 
    RSS.QUANTITE_J_REA AS "NB JOURS REA RSS" , 
    RSS.NB_JOUR_NEO1 AS "NB J NEO 1 RSS" , 
    RSS.NB_JOUR_NEO2 AS "NB J NEO 2 RSS" , 
    RSS.NB_JOUR_NEO3 AS "NB J NEO 3 RSS" , 
    RSS.NB_REP AS "NB J REA PED RSS" , 
    RSS.NB_JOUR_SOINS_INT AS "NB J SI RSS" , 
    RSS.NB_JOUR_SURV_CONT AS "NB J SURV CONT RSS" , 
    RSS.NB_JOUR_SOINS_INT_REA AS "NB J SI REA RSS" , 
    RSS.NB_JOUR_SOINS_PALL AS "NB J SOINS PALLIATIFS RSS" , 
    RSS.NB_ANTEPARTUM AS "NB J ANTEPARTUM RSS" , 
    RSS.NB_CAISSON_HB AS "NB CAISSON HYPERBARE RSS" , 
    RSS.NB_APHERESE AS "NB APHERESES RSS" , 
    RSS.NB_DIP AS "NB DIP RSS" , 
    RSS.NB_RDTH AS "NB RDTH RSS" , 
    RSS.NB_RDTH_PED AS "NB RDTH PED RSS" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN TB_UNITE_MEDICALE UMRSS
      ON  RSS.ID_UM_RETENU=UMRSS.ID_UNITE_MED

WHERE 
1=1 
 AND RSS.DATE_FIN >= '01/01/2026 00:00' AND RSS.DATE_FIN <= '31/12/2026 23:59'
