-- catalogue CORA : Analyse médico-administrative / Casemix / 04 - casemix DP
SELECT      RSS.DP_RETENU AS "DP Retenu" , 
    RSS.CODE_CMD || RSS.CODE_GHM AS "GHM" , 
    RTRIM(RTRIM(RSS.DP_RETENU)) || ' - ' || ((COALESCE((SELECT max(CIM.LIB)       FROM       CORA_CCAM.CIM_HIERARCHIE CIM WHERE UPPER(CIM.CODEFILS)=UPPER(RTRIM(RTRIM(RSS.DP_RETENU)))), 'Inconnu'))) AS "Libellé" , 
    UM.LIBELLE_UNITE_MED AS "LIBELLE UM RETENU" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RETENU" , 
    RSS.ID_RSS AS "RSS" , 
    RSS.DUREE_SEJOUR AS "NB Jours" , 
    RSS.AGE_ENTREE AS "Age" , 
    COALESCE((CASE WHEN RSS.MODE_SORTIE='9' THEN 1 ELSE 0 END),0) AS "Décès" , 
    COALESCE((CASE WHEN P.SEXE = 'M' THEN 1 END),0) AS "Masculin" , 
    COALESCE((CASE WHEN P.SEXE = 'F' THEN 1 END),0) AS "Féminin" , 
    COALESCE((CASE WHEN P.SEXE = 'F' THEN 0 WHEN P.SEXE = 'M' THEN 0 WHEN P.SEXE = '1' THEN 0 WHEN P.SEXE = '2' THEN 0 ELSE 1 END),0) AS "Sexe NR" , 
    (CASE WHEN RTRIM(RTRIM(RSS.BORNE_BASSE))='O' THEN 1 ELSE 0 END) AS "EXB" , 
    (CASE WHEN RSS.QUANTITE_J_DEP > 0 THEN 1 ELSE 0 END) AS "EXH" , 
    RSS.DATE_FIN AS "DATE" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN TB_PATIENT P
      ON  RSS.ID_PATIENT=P.ID_PATIENT
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUMX.ID_POLE, PRUMX.CODE_POLE, PRUMX.LIBELLE_POLE, PRUMX.ACTIF       FROM       TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUMX       WHERE       PUM2.ID_POLE=PRUMX.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )    AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU

WHERE 
1=1 
  AND RSS.CODE_CMD || RSS.CODE_GHM = 'Version 26' AND RSS.DATE_FIN >= '01/01/2026 00:00' AND RSS.DATE_FIN <= '31/12/2026 23:59'
