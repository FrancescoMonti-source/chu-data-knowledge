-- catalogue CORA : Analyse médico-administrative / CCAM / Hierarchie / CCAM - hierarchie
SELECT      RM.RANG AS "Rang" , 
    RM.LIBELLE AS "Libellé" , 
    COUNT(A.QUANTITE) AS "Nb Actes" , 
    AVG(RSS.DUREE_SEJOUR) AS "DMS" , 
    SUBSTR(NV.HIERARCHIE,1,6) AS "HIERARCHIE" , 
    RM.COD_MENU AS "CODE_MENU" 

FROM     TB_SYNTH_RSS RSS,
    TB_MVT M,
    TB_ACTE A,
    CORA_CCAM.CCAM_NV NV,
    CORA_CCAM_CNAM.R_MENU RM

WHERE 
1=1 
  AND RSS.ID_RSS=M.ID_RSS AND M.ID_MVT=A.ID_MVT AND A.CODE_ACTE=NV.CLE AND LPAD(to_char(RM.Rang), 6,'0') = SUBSTR(NV.HIERARCHIE,1,6) AND RSS.DATE_FIN>= '01/01/2026 00:00' AND RSS.DATE_FIN<= '31/12/2026 23:59' AND RM.COD_PERE = 1

 GROUP BY RM.COD_MENU, RM.RANG, RM.LIBELLE, SUBSTR(NV.HIERARCHIE,1,6)
 ORDER BY RM.RANG ASC
