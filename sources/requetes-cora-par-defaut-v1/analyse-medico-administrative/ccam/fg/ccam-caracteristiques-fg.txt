-- catalogue CORA : Analyse médico-administrative / CCAM / FG / CCAM - Caractéristiques FG
SELECT      FG.CARACTERISTIQUE AS "Code" , 
    CARA.LIBELLE AS "Libellé" , 
    COUNT(A.QUANTITE) AS "Nb Actes" , 
    AVG(RSS.DUREE_SEJOUR) AS "DMS" 

FROM     TB_SYNTH_RSS RSS,
    TB_MVT M,
    TB_ACTE A,
    CORA_CCAM.FG_CCAM_CARA FG,
    CORA_CCAM.FG_LIBCARA CARA

WHERE 
1=1 
  AND RSS.ID_RSS=M.ID_RSS AND M.ID_MVT=A.ID_MVT AND A.CODE_ACTE=FG.CODE_ACTE AND RTRIM(FG.CARACTERISTIQUE) = RTRIM(CARA.CARACTERISTIQUE) AND RSS.DATE_FIN>= '01/01/2026 00:00' AND RSS.DATE_FIN<= '31/12/2026 23:59' AND FG.CARACTERISTIQUE NOT IN ('39.1', '31.7', '16.7', '31.5', '32.5', '32.3', '22.2') AND FG.VERSION_FG = (SELECT max(VERSION_FG)  FROM  FG_LIBCARA) AND CARA.VERSION_FG = FG.VERSION_FG

 GROUP BY FG.CARACTERISTIQUE, CARA.LIBELLE
 ORDER BY LIBELLE ASC
