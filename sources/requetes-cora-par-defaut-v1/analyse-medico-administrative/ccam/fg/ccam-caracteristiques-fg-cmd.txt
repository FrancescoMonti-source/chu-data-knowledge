-- catalogue CORA : Analyse médico-administrative / CCAM / FG / CCAM - Caractéristiques FG - CMD
SELECT      RSS.CODE_CMD AS "Code CMD" , 
    RSS.CODE_GHM AS "Code GHM" , 
    COUNT(A.QUANTITE) AS "Nb Actes" , 
    AVG(RSS.DUREE_SEJOUR) AS "DMS" 

FROM     TB_SYNTH_RSS RSS,
    TB_MVT M,
    TB_ACTE A

WHERE 
1=1 
  AND RSS.ID_RSS=M.ID_RSS AND M.ID_MVT=A.ID_MVT AND RSS.DATE_FIN>= '01/01/2026 00:00' AND RSS.DATE_FIN<= '31/12/2026 23:59'

 GROUP BY RSS.CODE_CMD, RSS.CODE_GHM
 ORDER BY RSS.CODE_CMD ASC, RSS.CODE_GHM ASC
