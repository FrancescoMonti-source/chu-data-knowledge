-- catalogue CORA : Analyse médico-administrative / CCAM / FG / CCAM - Caractéristiques FG - Actes
SELECT      A.CODE_ACTE AS "Code Acte" , 
    NV.LIBELLE_LONG AS "Libellé" , 
    COUNT(A.QUANTITE) AS "Nb Actes" , 
    AVG(RSS.DUREE_SEJOUR) AS "DMS" 

FROM     TB_SYNTH_RSS RSS,
    TB_MVT M,
    TB_ACTE A,
    CORA_CCAM.FG_CCAM_CARA FG,
    CORA_CCAM.CCAM_NV NV

WHERE 
1=1 
  AND RSS.ID_RSS=M.ID_RSS AND M.ID_MVT=A.ID_MVT AND A.CODE_ACTE=FG.CODE_ACTE AND A.CODE_ACTE=NV.CLE AND RSS.DATE_FIN>= '01/01/2026 00:00' AND RSS.DATE_FIN<= '31/12/2026 23:59' AND FG.VERSION_FG= (SELECT MAX(VERSION_FG) FROM FG_LIBCARA)

 GROUP BY A.CODE_ACTE, NV.LIBELLE_LONG
 ORDER BY A.CODE_ACTE ASC, NV.LIBELLE_LONG ASC
