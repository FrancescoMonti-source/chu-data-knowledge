-- catalogue CORA : Analyse médico-administrative / Réadmission / readmission MVT
SELECT      P.IPP_PATIENT AS "NUM IPP PATIENT" , 
    P.NOM_USUEL_PATIENT AS "NOM USUEL" , 
    VueAdr.CODE_POSTAL AS "CODE POSTAL PAT" , 
    floor((S.DATE_ENTREE - P.DATE_NAIS)/365.25) AS "AGE PATIENT ANNEES ENT SEJ" , 
    URS.nom_util || ' ' || URS.prenom_util AS "MEDECIN RESP 1" , 
    URS.code_util AS "CODE RESP 1" , 
    S.NO_SEJOUR AS "NO SEJOUR 1" , 
    S.MODE_ENTREE AS "ME 1" , 
    S.CODE_PROVENANCE AS "CODE PROV 1" , 
    S.TYPE_SEJOUR AS "TYPE SEJOUR 1" , 
    S.DATE_SORTIE AS "DATE SORTIE 1" , 
    S.MODE_SORTIE AS "MODE SORTIE 1" , 
    S.CODE_DESTINATION AS "DEST 1" , 
    S.UF_ENTREE AS "UF ENTREE 1" , 
    S2.NO_SEJOUR AS "NO SEJOUR 2" , 
    S2.TYPE_SEJOUR AS "TYPE SEJOUR 2" , 
    S2.DATE_ENTREE AS "DATE ENTREE 2" , 
    S2.MODE_ENTREE AS "MODE ENTREE 2" , 
    S2.UF_ENTREE AS "UF ENTREE 2" , 
    S2.DATE_ENTREE-S.DATE_SORTIE AS "NB J READ" 

FROM     TB_PATIENT P,
    TB_SEJOUR S,
    TB_SEJOUR S2,
    UTIL URS,
    (SELECT ADR.ID_PATIENT AS ID_PAT_ADR, ADR.ADRESSE1, ADR.ADRESSE2, CADR.CODE_POSTAL, CADR.LIBELLE_COMMUNE, ADR.TEL_DOMICILE, ADR.TEL_PORTABLE, ADR.TEL_AUTRE FROM TB_PATIENT_ADRESSE ADR, TB_COMMUNE CADR WHERE CADR.ID_COMMUNE=ADR.ID_COMMUNE AND DEFAUT='O') VUEADR

WHERE 
1=1 
  AND P.ID_PATIENT=S.ID_PATIENT AND (S.TYPE_SEJOUR ='MCO' OR S.TYPE_SEJOUR ='HDJ' OR S.TYPE_SEJOUR ='URG' OR S.TYPE_SEJOUR ='EXT') AND S2.ID_PATIENT = S.ID_PATIENT AND S.ID_MEDECIN_RESP_SEJ=URS.id_util (+) AND P.ID_PATIENT=VueAdr.ID_PAT_ADR AND S2.NO_SEJOUR!=S.NO_SEJOUR AND S2.DATE_ENTREE<=(S.DATE_SORTIE+2) AND S2.DATE_ENTREE >=S.DATE_SORTIE AND S.DATE_SORTIE>= '01/01/2026 00:00' AND S.DATE_SORTIE<='31/12/2026 23:59' AND S.MODE_SORTIE='8'
