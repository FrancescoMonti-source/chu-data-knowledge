-- catalogue CORA : Analyse médico-administrative / Réadmission / readmission par pole ghm gp
SELECT      RSS.CODE_CMD || RSS.CODE_GHM AS "GHM RSS" , 
    RSS.CODE_CMD AS "CODE CMD RSS" , 
    SUBSTR(RSS.CODE_GHM,-1) AS "SUFFIXE CODE GHM RSS" , 
    PLGT.CODE_GHM_RACINE AS "RACINE GHM" , 
    UMRSS.CODE_UNITE_MED AS "CODE UNITE RETENUE RSS" , 
    PRUM.CODE_POLE AS "CODE POLE RUM" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RUM" , 
    RSS.DP_RETENU AS "DP RETENU" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    COUNT(RSS.ID_RSS) AS "RSS AVEC READ" , 
    LDA.LIB_PMSI_REGROUP AS "Libellé DA" , 
    LGA.LIB_PMSI_REGROUP AS "Libellé GA" , 
    LGP.LIB_PMSI_REGROUP AS "Libellé GP" , 
    S.NO_SEJOUR AS "NO SEJOUR 1" , 
    S2.NO_SEJOUR AS "NO SEJOUR 2" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN TB_SYNTH_RUM RUM
         INNER JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUM.ID_POLE, PRUM.CODE_POLE, PRUM.LIBELLE_POLE, PRUM.ACTIF  FROM  TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUM  WHERE  PUM2.ID_POLE=PRUM.ID_POLE) PRUM
          ON  RUM.ID_UNITE_MEDICALE=PRUM.ID_UNITE_MED
      ON  RSS.ID_RSS=RUM.ID_RSS
     LEFT JOIN PMSI_LIB_GHS_TARIF PLGT
      ON  RSS.CODE_CMD=PLGT.CMD AND RSS.CODE_GHS=PLGT.CODE_GHS AND RSS.CODE_GHM=PLGT.CODE_GHM AND RSS.DATE_FIN >= PLGT.DATE_DEBUT AND RSS.DATE_FIN<=PLGT.DATE_FIN
     LEFT JOIN PMSI_LIB_REGROUP LDA
      ON  PLGT.CODE_PMSI_REGROUP_DA=LDA.CODE_PMSI_REGROUP AND LDA.TYPE_PMSI_REGROUP='DA'
     LEFT JOIN PMSI_LIB_REGROUP LGA
      ON  PLGT.CODE_PMSI_REGROUP_GA=LGA.CODE_PMSI_REGROUP AND LGA.TYPE_PMSI_REGROUP='GA'
     LEFT JOIN PMSI_LIB_REGROUP LGP
      ON  PLGT.CODE_PMSI_REGROUP_GP=LGP.CODE_PMSI_REGROUP AND LGP.TYPE_PMSI_REGROUP='GP'
     INNER JOIN TB_UNITE_MEDICALE UMRSS
      ON  RSS.ID_UM_RETENU=UMRSS.ID_UNITE_MED
     INNER JOIN TB_SEJOUR S
      ON  RSS.ID_SEJOUR = S.ID_SEJOUR
     INNER JOIN TB_SEJOUR S2
      ON  S.ID_PATIENT = S2.ID_PATIENT AND S2.ID_SEJOUR!=S.ID_SEJOUR  AND S2.DATE_ENTREE >=S.DATE_SORTIE

WHERE 
1=1 
  AND (PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RUM.DATE_FIN_RUM) AND (PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RUM.DATE_FIN_RUM) AND RSS.DATE_FIN >= '01/01/2026 00:00' AND RSS.DATE_FIN <= '31/12/2026 23:59' AND S2.DATE_ENTREE<=(S.DATE_SORTIE+2)

 GROUP BY RSS.CODE_CMD || RSS.CODE_GHM, RSS.CODE_CMD, SUBSTR(RSS.CODE_GHM,-1), PLGT.CODE_GHM_RACINE, UMRSS.CODE_UNITE_MED, PRUM.CODE_POLE, PRUM.LIBELLE_POLE, RSS.DP_RETENU, RSS.DATE_FIN, LDA.LIB_PMSI_REGROUP, LGA.LIB_PMSI_REGROUP, LGP.LIB_PMSI_REGROUP, S.NO_SEJOUR, S2.NO_SEJOUR
