-- catalogue CORA : Analyse médico-administrative / Réadmission / readmission des sejours ayant eu un acte classant
SELECT      UM.LIBELLE_UNITE_MED AS "LIBELLE UM RETENU" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RETENU" , 
    RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    1 AS "NB READ" , 
    RSS.CODE_CMD || RSS.CODE_GHM AS "GHM RSS" , 
    RSS.CODE_CMD AS "CODE CMD RSS" , 
    A.CODE_ACTE AS "CODE ACTE" , 
    NV.LIBELLE_LONG AS "LIBELLE ACTE" , 
    UEA.CODE_UNITE AS "CODE UNITE EXEC" , 
    UEA.LIBELLE_UNITE AS "LIBELLEE UNITE EXEC" , 
    UOA.nom_util || ' ' || UOA.prenom_util AS "MED EXEC ACTIV" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUMX.ID_POLE, PRUMX.CODE_POLE, PRUMX.LIBELLE_POLE, PRUMX.ACTIF   FROM   TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUMX   WHERE   PUM2.ID_POLE=PRUMX.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )    AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU
     INNER JOIN TB_SEJOUR S
      ON  RSS.ID_SEJOUR=S.ID_SEJOUR
     INNER JOIN TB_SEJOUR S2
      ON  S2.ID_PATIENT = S.ID_PATIENT AND  S2.NO_SEJOUR!=S.NO_SEJOUR  AND S2.DATE_ENTREE >=S.DATE_SORTIE
     INNER JOIN TB_MVT M
      ON  RSS.ID_RSS=M.ID_RSS
     INNER JOIN TB_ACTE A
      ON  M.ID_MVT=A.ID_MVT
     INNER JOIN TB_DET_ACTE DA
      ON  A.ID_ACTE=DA.ID_ACTE
     INNER JOIN UTIL UOA
      ON  DA.ID_MEDECIN=UOA.ID_UTIL
     INNER JOIN FG_CCAM_CARA FG
      ON  A.CODE_ACTE=FG.CODE_ACTE AND FG.VERSION_FG= (SELECT MAX(VERSION_FG)   FROM   FG_LIBCARA)
     INNER JOIN CCAM_NV NV
      ON  A.CODE_ACTE=NV.CLE
     INNER JOIN TB_UNITE UEA
      ON  A.ID_UNITE_EXEC=UEA.ID_UNITE

WHERE 
1=1 
  AND DA.ID_ACTIVITE=1 AND FG.CARACTERISTIQUE = '0.0' AND (S.TYPE_SEJOUR ='MCO' OR S.TYPE_SEJOUR ='HDJ' OR S.TYPE_SEJOUR ='URG' OR S.TYPE_SEJOUR ='EXT') AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59' AND S2.DATE_ENTREE<=(S.DATE_SORTIE+2)
