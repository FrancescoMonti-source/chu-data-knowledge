-- catalogue CORA : Analyse médico-administrative / Réadmission / readmission dans les n jours en MCO
SELECT      RSS.CODE_CMD||RSS.CODE_GHM AS "GHM RSS" , 
    P.IPP_PATIENT AS "NUM IPP PATIENT" , 
    P.PRENOM_PATIENT AS "PRENOM" , 
    P.NOM_USUEL_PATIENT AS "NOM USUEL" , 
    P.NOM_NAISSANCE AS "NOM NAISSANCE" , 
    VueAdr.CODE_POSTAL AS "CODE POSTAL" , 
    floor((S.DATE_ENTREE - P.DATE_NAIS)/365.25) AS "AGE PATIENT" , 
    URS.nom_util || ' ' || URS.prenom_util AS "MEDECIN RESP 1" , 
    URS.code_util AS "CODE RESP 1" , 
    RSS.NO_SEJOUR AS "NO SEJOUR 1" , 
    S.MODE_ENTREE AS "ME 1" , 
    S.CODE_PROVENANCE AS "CP 1" , 
    RSS.DATE_FIN AS "FIN 1" , 
    RSS.MODE_SORTIE AS "MS 1" , 
    S.UF_ENTREE AS "UF ENTREE 1" , 
    RSS.CODE_DESTINATION AS "DEST 1" , 
    S2.NO_SEJOUR AS "NO SEJOUR 2" , 
    S2.DATE_ENTREE AS "ENTREE 2" , 
    S2.MODE_ENTREE AS "ME 2" , 
    UMRSS.CODE_UNITE_MED AS "CODE UNITE RETENUE RSS" , 
    UMRSS.LIBELLE_UNITE_MED AS "LIBELLE UNITE  RETENUE RSS" , 
    S2.UF_ENTREE AS "UF ENTREE 2" , 
    S2.DATE_ENTREE-S.DATE_SORTIE AS "NB J READ" 

FROM     TB_PATIENT P
     LEFT JOIN (SELECT ADR.ID_PATIENT AS ID_PAT_ADR, ADR.ADRESSE1, ADR.ADRESSE2, CADR.CODE_POSTAL, CADR.LIBELLE_COMMUNE, ADR.TEL_DOMICILE, ADR.TEL_PORTABLE, ADR.TEL_AUTRE             FROM             TB_PATIENT_ADRESSE ADR, TB_COMMUNE CADR             WHERE             CADR.ID_COMMUNE=ADR.ID_COMMUNE AND DEFAUT='O') VUEADR
      ON  P.ID_PATIENT=VueAdr.ID_PAT_ADR
     INNER JOIN TB_SEJOUR S
         INNER JOIN TB_SYNTH_RSS RSS
             LEFT JOIN TB_UNITE_MEDICALE UMRSS
              ON  RSS.ID_UM_RETENU=UMRSS.ID_UNITE_MED
          ON  S.ID_SEJOUR=RSS.ID_SEJOUR
      ON  P.ID_PATIENT=S.ID_PATIENT
     LEFT JOIN CORA_UTIL.UTIL URS
      ON  S.ID_MEDECIN_RESP_SEJ=URS.id_util
     INNER JOIN TB_SEJOUR S2
      ON  P.ID_PATIENT=S2.ID_PATIENT

WHERE 
1=1 
  AND S2.NO_SEJOUR!=S.NO_SEJOUR AND S2.DATE_ENTREE<=(S.DATE_SORTIE+2) AND S2.DATE_ENTREE >=S.DATE_SORTIE AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59' AND RSS.MODE_SORTIE='8'
