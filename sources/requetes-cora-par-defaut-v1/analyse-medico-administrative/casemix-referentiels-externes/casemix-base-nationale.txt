-- catalogue CORA : Analyse médico-administrative / Casemix référentiels externes / casemix - base nationale
SELECT      RSS.CODE_CMD AS "CMD" , 
    RSS.CODE_CMD  || ' - ' || LIB.LIB_CMD AS "Libellé CMD" , 
    RSS.CODE_CMD || SUBSTR(RSS.CODE_GHM,1,3) AS "Racine GHM" , 
    RSS.CODE_CMD || SUBSTR(RSS.CODE_GHM,1,3)  || ' - ' || LIBR.LIBELLE_RACINE AS "Libellé Racine GHM" , 
    COUNT(RSS.ID_RSS) AS "Nb RSS Local" , 
    SUM(RSS.DUREE_SEJOUR) AS "Somme Durees Local" , 
    SUM(RSS.AGE_ENTREE) AS "Somme Ages Local" , 
    COALESCE(SUM(CASE WHEN RSS.MODE_SORTIE='9' THEN 1 ELSE 0 END),0) AS "NB Décès Local" , 
    SUM(CASE WHEN RTRIM(RSS.BORNE_BASSE)='O' THEN 1 ELSE 0 END) AS "Nb EXB Local" , 
    SUM(CASE WHEN RSS.QUANTITE_J_DEP > 0 THEN 1 ELSE 0 END) AS "NB EXH Local" , 
    SUM(EX.NB_RSS_BN) AS "Nb RSS BN" , 
    SUM(EX.SUM_DUREE_BN) AS "Somme Durees BN" , 
    SUM(EX.SUM_AGE_BN) AS "Somme Ages BN" , 
    SUM(EX.SORTIE_9_BN) AS "NB Décès BN" , 
    SUM(EX.NB_EXB_BN) AS "Nb EXB BN" , 
    SUM(EX.NB_EXH_BN) AS "NB EXH BN" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN PMSI_LIB_CMD LIB
      ON  UPPER(LIB.CODE_CMD)=RSS.CODE_CMD
     INNER JOIN FG_RGHM_INFO LIBR
      ON  LIBR.RACINE_GHM=RSS.CODE_CMD || SUBSTR(RSS.CODE_GHM,1,3) AND LIBR.VERSION_FG=(SELECT MAX(VERSION_FG)       FROM       FG_RGHM_INFO LIBR2)
     INNER JOIN TB_SEJOUR S
         INNER JOIN TB_PATIENT P
          ON  S.ID_PATIENT=P.ID_PATIENT
      ON  RSS.ID_SEJOUR=S.ID_SEJOUR
     INNER JOIN (SELECT BNA.CODE_CMD, BNA.CODE_GHM_RACINE || BNA.SEVERITE AS "CODE_GHM_BN", SUM(BNA.SORTIE_9) AS "SORTIE_9_BN", SUM(BNA.NB_EXB) AS "NB_EXB_BN", SUM(BNA.NB_EXH) AS "NB_EXH_BN", SUM(BNA.NB_SEJOUR) AS "NB_RSS_BN", SUM(BNA.AGE_ANNEE) AS "SUM_AGE_BN", SUM(BNA.DUREE_TOTALE) AS "SUM_DUREE_BN"       FROM       CORA_CCAM.BNA_EXPLOITATION BNA GROUP BY BNA.CODE_CMD, BNA.CODE_GHM_RACINE || BNA.SEVERITE) EX
      ON  EX.CODE_CMD=RSS.CODE_CMD AND EX.CODE_GHM_BN=RSS.CODE_GHM

WHERE 
1=1 
  AND RSS.DATE_FIN>='01/01/2025 00:00' AND RSS.DATE_FIN<='31/12/2025 23:59'

 GROUP BY RSS.CODE_CMD, RSS.CODE_CMD || SUBSTR(RSS.CODE_GHM,1,3), RSS.CODE_CMD  || ' - ' || LIB.LIB_CMD, RSS.CODE_CMD || SUBSTR(RSS.CODE_GHM,1,3)  || ' - ' || LIBR.LIBELLE_RACINE, RSS.ID_RSS
 ORDER BY RSS.CODE_CMD ASC, RSS.CODE_CMD || SUBSTR(RSS.CODE_GHM,1,3) ASC
