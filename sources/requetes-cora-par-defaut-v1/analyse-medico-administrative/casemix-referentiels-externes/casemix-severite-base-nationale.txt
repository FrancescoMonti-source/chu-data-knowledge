-- catalogue CORA : Analyse médico-administrative / Casemix référentiels externes / casemix severite - base nationale
SELECT      EX.CODE_CMD||EX.CODE_GHM_RACINE AS "GHM" , 
    RSS.CODE_CMD || RSS.CODE_GHM || ' - '|| coalesce((select PMSI.LIBELLE_GHS    FROM    PMSI_LIB_GHS_TARIF PMSI  where PMSI.GHS=RSS.CODE_CMD||RSS.CODE_GHM AND PMSI.GET_LIBELLE='O' and rownum=1), 'ghm inconnu') AS "Libellé" , 
    CASE WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = '1' THEN '1' WHEN SUBSTR(CODE_GHM, 4 ,1) = '2' THEN '2'  WHEN SUBSTR(CODE_GHM, 4 ,1) = '3' THEN '3' WHEN SUBSTR(CODE_GHM, 4 ,1) = '4' THEN '4'  WHEN SUBSTR(CODE_GHM, 4 ,1) = 'A' OR SUBSTR(CODE_GHM, 4 ,1) = 'B' OR SUBSTR(CODE_GHM, 4 ,1) = 'C'  OR SUBSTR(CODE_GHM, 4 ,1) = 'D' OR SUBSTR(CODE_GHM, 4 ,1) = 'E' THEN 'A, B, C, D, E'  WHEN SUBSTR(CODE_GHM, 4 ,1) = 'J' THEN 'J' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'S' THEN 'S'  WHEN SUBSTR(CODE_GHM, 4 ,1) = 'T' THEN 'T' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'V' THEN 'V'  WHEN SUBSTR(CODE_GHM, 4 ,1) = 'W' THEN 'W' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'Z' AND RSS.CODE_CMD <> '28'  THEN 'Z' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'Z' AND RSS.CODE_CMD = '28' THEN 'Séances' ELSE 'Inconnu'  END AS "Niveau" , 
    RSS.ID_RSS AS "Nb RSS" , 
    (EX.NB_SEJOUR) AS "Nb Séjour ATIH" 

FROM     TB_SYNTH_RSS RSS,
    (SELECT CODE_CMD,CODE_GHM_RACINE, RTRIM(CODE_GHM_RACINE)||LTRIM(SEVERITE) AS GHM,SEVERITE,SUM(NB_SEJOUR) AS "NB_SEJOUR",SUM(DUREE_TOTALE) AS "DUREE_SEJOUR_AT"     FROM    CORA_CCAM.BNA_EXPLOITATION GROUP BY CODE_CMD,CODE_GHM_RACINE,SEVERITE) EX

WHERE 
1=1 
  AND EX.CODE_CMD=RSS.CODE_CMD AND RSS.DATE_FIN>='01/01/2025 00:00' AND RSS.DATE_FIN<='31/12/2025 23:59' AND lTRIM(EX.GHM)=lTRIM(RSS.CODE_GHM) AND RSS.ID_RSS = (SELECT MAX(RSS2.ID_RSS)    FROM    TB_SYNTH_RSS RSS2    WHERE    RSS2.ID_SEJOUR = RSS.ID_SEJOUR)

 GROUP BY RSS.CODE_GHM, EX.CODE_CMD, CODE_GHM_RACINE, EX.GHM, RSS.CODE_CMD, CASE WHEN SUBSTR(CODE_GHM, 4 ,1) = '1'   THEN '1' WHEN SUBSTR(CODE_GHM, 4 ,1) = '2' THEN '2'   WHEN SUBSTR(CODE_GHM, 4 ,1) = '3' THEN '3'   WHEN SUBSTR(CODE_GHM, 4 ,1) = '4'   THEN '4' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'A' OR   SUBSTR(CODE_GHM, 4 ,1) = 'B' OR   SUBSTR(CODE_GHM, 4 ,1) = 'C' OR SUBSTR(CODE_GHM, 4 ,1) = 'D' OR     SUBSTR(CODE_GHM, 4 ,1) = 'E' THEN 'A, B, C, D, E' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'J' THEN 'J'   WHEN   SUBSTR(CODE_GHM, 4 ,1) = 'S' THEN 'S' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'T' THEN 'T'   WHEN       SUBSTR(CODE_GHM, 4 ,1) = 'V' THEN 'V' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'W' THEN 'W'   WHEN    SUBSTR(CODE_GHM, 4 ,1) = 'Z' AND RSS.CODE_CMD <> '28' THEN 'Z'   WHEN SUBSTR(CODE_GHM, 4 ,1)    = 'Z' AND RSS.CODE_CMD = '28' THEN 'Séances' ELSE 'Inconnu' END, NB_SEJOUR, DUREE_SEJOUR_AT, RSS.ID_RSS
 ORDER BY GHM ASC
