-- catalogue CORA : Analyse médico-administrative / Domaine d'activité / regroupement GP - analyse SRC REA NNx
SELECT      RSS.ID_RSS AS "RSS" , 
    P.code_pmsi_regroup_gp AS "Code GP" , 
    L.LIB_PMSI_REGROUP AS "Libellé" , 
    CASE WHEN RSS.DUREE_SEJOUR=0 THEN 1 ELSE RSS.DUREE_SEJOUR END AS "Nb jour" , 
    RSS.MNT_TOTAL AS "Mnt TOTAL" , 
    RSS.MONTANT_GHS AS "Mnt GHS" , 
    RSS.MONTANT_DEP AS "Mnt EXH" , 
    RSS.QUANTITE_J_DEP AS "Nb EXH" , 
    RSS.MONTANT_REDU AS "Mnt EXB" , 
    RSS.QUANTITE_J_REA AS "Nb REA" , 
    RSS.MONTANT_REA AS "Mnt REA" , 
    RSS.NB_JOUR_SOINS_INT_REA AS "Nb SI REA" , 
    RSS.MNT_JOUR_SOINS_INT_REA AS "Mnt SI REA" , 
    RSS.NB_JOUR_SOINS_INT AS "Nb STF" , 
    RSS.MNT_JOUR_SOINS_INT AS "Mnt STF" , 
    RSS.NB_JOUR_SURV_CONT AS "Nb SRC" , 
    RSS.MNT_JOUR_SURV_CONT AS "Mnt SRC" , 
    RSS.NB_JOUR_NEO1 AS "Nb NN1" , 
    RSS.MNT_JOUR_NEO1 AS "Mnt NN1" , 
    RSS.NB_JOUR_NEO2 AS "Nb NN2" , 
    RSS.MNT_JOUR_NEO2 AS "Mnt NN2" , 
    RSS.NB_JOUR_NEO3 AS "Nb NN3" , 
    RSS.MNT_JOUR_NEO3 AS "Mnt NN3" , 
    RSS.NB_REP AS "Nb REP" , 
    RSS.MNT_REP AS "Mnt REP" , 
    RSS.NB_CAISSON_HB AS "Nb CHB" , 
    RSS.MNT_CAISSON_HB AS "Mnt CHB" , 
    RSS.NB_APHERESE AS "Nb APH" , 
    RSS.MNT_APHERESE AS "Mnt APH" , 
    RSS.NB_DIP AS "Nb DIP" , 
    RSS.MNT_DIP AS "Mnt DIP" , 
    RSS.NB_DIALYSE AS "Nb DIA" , 
    RSS.MT_DIALYSE AS "Mnt DIA" , 
    RSS.NB_RDTH AS "Nb RDTH" , 
    RSS.MNT_RDTH AS "Mnt RDTH" , 
    RSS.NB_ANTEPARTUM AS "Nb ANT" , 
    RSS.MNT_ANTEPARTUM AS "Mnt ANT" , 
    RSS.NB_RDTH_PED AS "Nb RDTH PED" , 
    RSS.MNT_RDTH_PED AS "Mnt RDTH PED" , 
    UM.LIBELLE_UNITE_MED AS "LIBELLE UM RETENU" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RETENU" , 
    RSS.DATE_FIN AS "DATE" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUMX.ID_POLE, PRUMX.CODE_POLE, PRUMX.LIBELLE_POLE, PRUMX.ACTIF     FROM     TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUMX     WHERE     PUM2.ID_POLE=PRUMX.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )  AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU,
    PMSI_LIB_GHS_TARIF P,
    PMSI_LIB_REGROUP L

WHERE 
1=1 
  AND RSS.CODE_CMD=P.CMD AND RSS.CODE_GHM=P.CODE_GHM AND RSS.DATE_FIN>=P.DATE_DEBUT AND RSS.DATE_FIN<=P.DATE_FIN AND P.GET_LIBELLE='O' AND P.CODE_PMSI_REGROUP_GP=L.CODE_PMSI_REGROUP AND L.TYPE_PMSI_REGROUP='GP' AND RSS.DATE_FIN>='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
