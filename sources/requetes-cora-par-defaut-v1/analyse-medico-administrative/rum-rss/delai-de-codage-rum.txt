-- catalogue CORA : Analyse médico-administrative / RUM-RSS / délai de codage RUM
SELECT      TB_SYNTH_RUM.ID_RUM AS "NB RUM" , 
    TB_SEJOUR.NO_SEJOUR AS "NO_SEJOUR" , 
    TB_DIAG.CODE_DIAG AS "CODE_DIAG" , 
    TB_DIAG.TYPE_DIAG AS "TYPE_DIAG" , 
    TB_PATIENT.IPP_PATIENT AS "IPP_PATIENT" , 
    UM_RUM.CODE_UNITE_MED AS "CODE_UNITE_MED" , 
    TB_DIAG.DATE_CREAT AS "DATE_CREAT" , 
    TB_MVT.DATE_FIN_MVT AS "DATE FIN MVT" , 
    TB_SYNTH_RUM.DATE_DEBUT_RUM AS "DATE_DEBUT_RUM" , 
    TB_SYNTH_RUM.DATE_FIN_RUM AS "DATE_FIN_RUM" , 
    floor( TB_DIAG.DATE_CREAT-TB_MVT.DATE_FIN_MVT) AS "DELAI" 

FROM     TB_MVT ,
    TB_SEJOUR ,
    TB_PATIENT ,
    TB_DIAG ,
    TB_SYNTH_RUM ,
    TB_UNITE_MEDICALE UM_RUM,
    TB_MVT_RUM TB_MVT_RUM2

WHERE 
1=1 
  AND TB_SEJOUR.ID_PATIENT=TB_PATIENT.ID_PATIENT AND TB_MVT.ID_SEJOUR=TB_SEJOUR.ID_SEJOUR AND TB_MVT.ID_MVT=TB_DIAG.ID_MVT AND UM_RUM.ID_UNITE_MED=TB_SYNTH_RUM.ID_UNITE_MEDICALE AND TB_MVT.ID_MVT=TB_MVT_RUM2.ID_MVT AND TB_MVT_RUM2.ID_RUM=TB_SYNTH_RUM.ID_RUM AND TB_SEJOUR.DATE_SORTIE >='01/01/2026 00:00' AND TB_SEJOUR.DATE_SORTIE  <='31/12/2026 23:59'
