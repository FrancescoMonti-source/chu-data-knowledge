-- catalogue CORA : Analyse médico-administrative / RUM-RSS / RSS UHCD  au moins 2 jours
SELECT      RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    RSS.DUREE_SEJOUR AS "DUREE RSS" , 
    UMVT.TYPE_AUTORISATION AS "UNITE MVT AUTO SROS" , 
    M.ID_MVT AS "ID_MVT" 

FROM     TB_MVT_RUM MVTRUM,
    TB_MVT M,
    TB_SYNTH_RSS RSS,
    TB_SYNTH_RUM RUM,
    TB_UNITE UMVT

WHERE 
1=1 
  AND RSS.ID_RSS=RUM.ID_RSS AND RUM.ID_RUM=MVTRUM.ID_RUM AND MVTRUM.ID_MVT=M.ID_MVT AND M.ID_UNITE_M=UMVT.CODE_UNITE AND UMVT.TYPE_AUTORISATION  like '07%' AND RSS.DUREE_SEJOUR >1 AND RSS.NB_RUM =1 AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
