-- catalogue CORA : Analyse médico-administrative / RUM-RSS / codes erreurs (DIM ROUEN)
SELECT      CASE WHEN L.BLOQUANT='O' THEN 'Bloquant' ELSE CASE WHEN etn.ID IS NOT NULL THEN 'Non Bloquant Eteint' ELSE 'Non Bloquant' END END AS "Bloquant" , 
    LPAD(RTRIM(LTRIM(err_rss.str)),3,'0') AS "Code Retour" , 
    L.CODE AS "Code Erreur" , 
    UM.LIBELLE_UNITE_MED AS "LIBELLE UM RETENU" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RETENU" , 
    L.LIB_ERR AS "Libellé" , 
    RSS.ID_RSS AS "RSS" , 
    RSS.DATE_FIN AS "DATE" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN (select distinct regexp_SUBSTR(list_err,'[^,]+',1,1) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,2) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,3) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,4) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,5) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,6) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,7) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,8) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,9) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,10) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,11) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,12) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,13) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,14) as str, id_rss          FROM          tb_synth_rss UNION select regexp_SUBSTR(list_err,'[^,]+',1,15) as str, id_rss          FROM          tb_synth_rss) err_rss
      ON  err_rss.id_rss=rss.id_rss
     LEFT JOIN PMSI_LIB_ERREUR L
      ON  LPAD(RTRIM(LTRIM(err_rss.str)),3,'0')=L.CODE_ERREUR
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUMX.ID_POLE, PRUMX.CODE_POLE, PRUMX.LIBELLE_POLE, PRUMX.ACTIF             FROM             TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUMX             WHERE             PUM2.ID_POLE=PRUMX.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )    AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU
     LEFT JOIN TB_RSS_ERREUR_DESACTIVE etn
      ON  etn.ID_RSS = rss.ID_RSS AND etn.CODE_ERREUR = LPAD(RTRIM(LTRIM(err_rss.str)),3,'0')

WHERE 
1=1 
  AND RSS.DATE_FIN>='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59' AND etn.CODE_ERREUR is NULL
