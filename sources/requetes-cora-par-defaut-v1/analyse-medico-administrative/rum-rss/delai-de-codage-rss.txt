-- catalogue CORA : Analyse médico-administrative / RUM-RSS / délai de codage RSS
SELECT      RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSSS.ETAT_GROUPAGE AS "RSS - SUIVI ETAT" , 
    RSSS.DATE_INFO AS "DATE GRP" , 
    floor(  RSSS.DATE_INFO- RSS.DATE_FIN) AS "DELAI GRP" , 
    RSS.ID_RSS AS "Nb RSS" 

FROM     TB_SYNTH_RSS RSS,
    TB_RSS_SUIVI RSSS

WHERE 
1=1 
  AND RSSS.ID_RSS=RSS.ID_RSS AND RSSS.ETAT_GROUPAGE='P' AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN  <='31/12/2026 23:59'
