-- catalogue CORA : Analyse médico-administrative / RUM-RSS / analyse des RUM par UM - copie
SELECT      1 AS "NB  RUM" , 
    NVL(UMAXEPMSI.CODE_UNITE_MED, 'N.R.') AS "CODE UM AXE PRINC" , 
    NVL(UMAXEPMSI.LIBELLE_UNITE_MED, 'N.R.') AS "LIBELLE UM AXE PRINC" , 
    NVL(UMAXECOMPL.CODE_UNITE_MED, 'N.R.') AS "CODE UM AXE COMPL" , 
    NVL(UMAXECOMPL.LIBELLE_UNITE_MED, 'N.R.') AS "LIBELLE UM AXE COMPL" , 
    RUM.DUREE AS "DUREE RUM" , 
    RUM.DATE_DEBUT_RUM AS "DATE DEBUT RUM" , 
    RUM.DATE_FIN_RUM AS "DATE FIN RUM" , 
    CIM.LIB AS "LIBELLE DP" , 
    RUM.DIAG_P AS "CODE DP" 

FROM     TB_SYNTH_RUM RUM
     LEFT JOIN TB_UNITE_MEDICALE UMAXEPMSI
      ON  RUM.ID_UNITE_MEDICALE= UMAXEPMSI.ID_UNITE_MED
     LEFT JOIN TB_MVT_COMPL AXECOMPL
      ON  RUM.ID_SEJOUR= AXECOMPL.ID_SEJOUR  AND    RUM.DATE_DEBUT_RUM >= AXECOMPL.DATE_DEBUT AND RUM.DATE_DEBUT_RUM < AXECOMPL.DATE_FIN
     LEFT JOIN TB_UNITE_M_UNITE UMUFAXECOMPL
      ON  AXECOMPL.ID_UNITE_PMSI= UMUFAXECOMPL.ID_UNITE AND ((UMUFAXECOMPL.DATE_DEBUT IS NULL OR UMUFAXECOMPL.DATE_DEBUT<=RUM.DATE_FIN_RUM) AND (UMUFAXECOMPL.DATE_FIN IS NULL OR UMUFAXECOMPL.DATE_FIN>=RUM.DATE_FIN_RUM))
     LEFT JOIN TB_UNITE_MEDICALE UMAXECOMPL
      ON  UMUFAXECOMPL.ID_UNITE_MED= UMAXECOMPL.ID_UNITE_MED
     LEFT JOIN CORA_CCAM.CIM_HIERARCHIE CIM
      ON  RUM.DIAG_P=CIM.CODEFILS

WHERE 
1=1 
  AND RUM.DATE_FIN_RUM >='01/01/2026 00:00' AND RUM.DATE_FIN_RUM <='31/12/2026 23:59'
