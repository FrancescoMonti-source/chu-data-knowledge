-- catalogue CORA : Analyse médico-administrative / RUM-RSS / analyse detaillée des RUM multi unités
SELECT      RSS.ID_RSS AS "ID RSS" , 
    RSS.NB_RUM AS "NB DE RUM" , 
    RSS.NO_SEJOUR AS "N° SEJOUR" , 
    UMHEBRUM.CODE_UNITE_MED AS "UM HEB RUM 1" , 
    UMHEBRUM.LIBELLE_UNITE_MED AS "LIB UM HEB RUM 1" , 
    UMRESPRUM.CODE_UNITE_MED AS "UM RESP RUM 1" , 
    UMRESPRUM.LIBELLE_UNITE_MED AS "LIB UM RESP RUM 1" , 
    RUM.DUREE AS "DUREE RUM 1" , 
    RUM.DATE_DEBUT_RUM AS "DEBUT RUM 1" , 
    RUM.DATE_FIN_RUM AS "FIN RUM 1" , 
    CIM.LIB AS "LIB DIAG 1" , 
    D.CODE_DIAG AS "DIAG 1" , 
    UMHEBRUM2.CODE_UNITE_MED AS "UM HEB RUM 2" , 
    UMHEBRUM2.LIBELLE_UNITE_MED AS "LIB UM HEB RUM 2" , 
    UMRESPRUM.CODE_UNITE_MED AS "UM RESP RUM 2" , 
    UMRESPRUM2.LIBELLE_UNITE_MED AS "LIB UM RESP RUM 2" , 
    RUM2.DUREE AS "DUREE RUM 2" , 
    RUM2.DATE_DEBUT_RUM AS "DEBUT RUM 2" , 
    RUM2.DATE_FIN_RUM AS "FIN RUM 2" , 
    CIM2.LIB AS "LIB DIAG 2" , 
    D2.CODE_DIAG AS "DIAG 2" , 
    RSS.ID_RSS AS "NB RSS" 

FROM     TB_SYNTH_RSS RSS,
    TB_SYNTH_RUM RUM
     LEFT JOIN TB_UNITE_MEDICALE UMHEBRUM
      ON  RUM.ID_UM_HEBERG= UMHEBRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UMRESPRUM
      ON  RUM.ID_UM_RESP_MED= UMRESPRUM.ID_UNITE_MED,
    TB_MVT_RUM MVTRUM,
    TB_MVT M,
    TB_DIAG D,
    CORA_CCAM.CIM_HIERARCHIE CIM,
    TB_SYNTH_RUM RUM2
     LEFT JOIN TB_UNITE_MEDICALE UMHEBRUM2
      ON  RUM2.ID_UM_HEBERG= UMHEBRUM2.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UMRESPRUM2
      ON  RUM2.ID_UM_RESP_MED= UMRESPRUM2.ID_UNITE_MED,
    TB_MVT_RUM MVTRUM2,
    TB_MVT M2,
    TB_DIAG D2,
    CORA_CCAM.CIM_HIERARCHIE CIM2

WHERE 
1=1 
  AND RSS.NB_RUM>1 AND RSS.ID_RSS=RUM.ID_RSS AND RUM.ID_RUM=MVTRUM.ID_RUM AND MVTRUM.ID_MVT=M.ID_MVT AND M.ID_MVT=D.ID_MVT AND D.TYPE_DIAG = 'P' AND D.CODE_DIAG=CIM.CODEFILS AND RUM2.ID_RSS=RUM.ID_RSS AND RUM2.ID_RUM!=RUM.ID_RUM AND RUM2.DATE_DEBUT_RUM=RUM.DATE_FIN_RUM AND RUM2.DATE_DEBUT_RUM >=RUM.DATE_FIN_RUM AND RUM2.ID_RUM=MVTRUM2.ID_RUM AND MVTRUM2.ID_MVT=M2.ID_MVT AND M2.ID_MVT=D2.ID_MVT AND D2.TYPE_DIAG = 'P' AND D2.CODE_DIAG=CIM2.CODEFILS AND RUM.DATE_FIN_RUM >='01/01/2026 00:00' AND RUM.DATE_FIN_RUM <='31/12/2026 23:59'
