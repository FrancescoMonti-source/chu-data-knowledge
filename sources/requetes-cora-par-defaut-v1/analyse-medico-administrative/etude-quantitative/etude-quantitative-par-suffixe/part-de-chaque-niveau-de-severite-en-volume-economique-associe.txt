-- catalogue CORA : Analyse médico-administrative / Etude quantitative / etude quantitative par suffixe / part de chaque niveau de severité en volume economique associé
SELECT      UM.LIBELLE_UNITE_MED AS "LIBELLE UM RETENU" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RETENU" , 
    CASE WHEN SUBSTR(CODE_GHM, 4 ,1) = 'J' OR SUBSTR(CODE_GHM, 4 ,1) = 'T' THEN 'Niv J/T' WHEN SUBSTR(CODE_GHM, 4 ,1) = '1' OR SUBSTR(CODE_GHM, 4 ,1) = 'A' THEN 'Niv 1/A' WHEN SUBSTR(CODE_GHM, 4 ,1) = '2' OR SUBSTR(CODE_GHM, 4 ,1) = 'B' THEN 'Niv 2/B' WHEN SUBSTR(CODE_GHM, 4 ,1) = '3' OR SUBSTR(CODE_GHM, 4 ,1) = 'C' THEN 'Niv 3/C' WHEN SUBSTR(CODE_GHM, 4 ,1) = '4' OR SUBSTR(CODE_GHM, 4 ,1) = 'D' THEN 'Niv 4' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'Z' AND CODE_CMD <> '28' THEN 'Niv Z' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'Z' AND CODE_CMD = '28' THEN 'Séances' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'S' THEN 'Niv S' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'T' THEN 'Niv T' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'V' THEN 'Niv V' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'W'  THEN 'Niv W' WHEN SUBSTR(CODE_GHM, 4 ,1) = 'E'  THEN 'Niv E/Décés'  ELSE 'Inconnu' END AS "Groupe Sévérité" , 
    SUBSTR(CODE_GHM, 4 ,1) AS "Niveau Sévérité" , 
    RSS.ID_RSS AS "Nb RSS" , 
    RSS.DUREE_SEJOUR AS "Durée" , 
    RSS.MNT_TOTAL AS "Mnt Total" , 
    RSS.DATE_FIN AS "Date Fin RSS" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUMX.ID_POLE, PRUMX.CODE_POLE, PRUMX.LIBELLE_POLE, PRUMX.ACTIF    FROM    TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUMX    WHERE    PUM2.ID_POLE=PRUMX.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )    AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU

WHERE 
1=1 
  AND RSS.ID_RSS = (SELECT MAX(RSS2.ID_RSS)    FROM    TB_SYNTH_RSS RSS2    WHERE    RSS2.ID_SEJOUR = RSS.ID_SEJOUR) AND RSS.DATE_FIN >= '01/01/2026 00:00' AND RSS.DATE_FIN <= '31/12/2026 23:59'
