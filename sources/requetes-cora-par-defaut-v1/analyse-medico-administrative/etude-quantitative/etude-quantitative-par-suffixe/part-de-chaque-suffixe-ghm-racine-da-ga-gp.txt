-- catalogue CORA : Analyse médico-administrative / Etude quantitative / etude quantitative par suffixe / part de chaque suffixe - GHM racine DA GA GP
SELECT      UM.LIBELLE_UNITE_MED AS "LIBELLE UM RETENU" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RETENU" , 
    CASE WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = 'J' OR SUBSTR(RSS.CODE_GHM, 4 ,1) = 'T' THEN 'Niv J/T' WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = '1' OR SUBSTR(RSS.CODE_GHM, 4 ,1) = 'A' THEN 'Niv 1/A' WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = '2' OR SUBSTR(RSS.CODE_GHM, 4 ,1) = 'B' THEN 'Niv 2/B' WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = '3' OR SUBSTR(RSS.CODE_GHM, 4 ,1) = 'C' THEN 'Niv 3/C' WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = '4' OR SUBSTR(RSS.CODE_GHM, 4 ,1) = 'D' THEN 'Niv 4' WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = 'Z' AND CODE_CMD <> '28' THEN 'Niv Z' WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = 'Z' AND CODE_CMD = '28' THEN 'Séances' WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = 'S' THEN 'Niv S' WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = 'T' THEN 'Niv T' WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = 'V' THEN 'Niv V' WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = 'W' THEN 'Niv W'  WHEN SUBSTR(RSS.CODE_GHM, 4 ,1) = 'E' THEN 'Niv E/Décés'  ELSE 'Inconnu' END AS "Groupe Sévérité" , 
    SUBSTR(RSS.CODE_GHM, 4 ,1) AS "Niveau Sévérité" , 
    RSS.ID_RSS AS "Nb RSS" , 
    RSS.DUREE_SEJOUR AS "Durée" , 
    RSS.MNT_TOTAL AS "Mnt Total" , 
    RSS.DATE_FIN AS "Date Fin RSS" , 
    NVL(PLGT.CODE_GHM_RACINE,'Inconnu') AS "Racine GHM" , 
    NVL(LDA.LIB_PMSI_REGROUP,'Inconnu') AS "Libellé DA" , 
    NVL(LGA.LIB_PMSI_REGROUP,'Inconnu') AS "Libellé GA" , 
    NVL(LGP.LIB_PMSI_REGROUP,'Inconnu') AS "Libellé GP" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN PMSI_LIB_GHS_TARIF PLGT
      ON  RSS.CODE_CMD=PLGT.CMD AND RSS.CODE_GHS=PLGT.CODE_GHS AND RSS.CODE_GHM=PLGT.CODE_GHM AND RSS.DATE_FIN >= PLGT.DATE_DEBUT AND RSS.DATE_FIN<=PLGT.DATE_FIN
     LEFT JOIN PMSI_LIB_REGROUP LDA
      ON  PLGT.CODE_PMSI_REGROUP_DA=LDA.CODE_PMSI_REGROUP AND LDA.TYPE_PMSI_REGROUP='DA'
     LEFT JOIN PMSI_LIB_REGROUP LGA
      ON  PLGT.CODE_PMSI_REGROUP_GA=LGA.CODE_PMSI_REGROUP AND LGA.TYPE_PMSI_REGROUP='GA'
     LEFT JOIN PMSI_LIB_REGROUP LGP
      ON  PLGT.CODE_PMSI_REGROUP_GP=LGP.CODE_PMSI_REGROUP AND LGP.TYPE_PMSI_REGROUP='GP'
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUMX.ID_POLE, PRUMX.CODE_POLE, PRUMX.LIBELLE_POLE, PRUMX.ACTIF    FROM    TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUMX    WHERE    PUM2.ID_POLE=PRUMX.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )    AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU

WHERE 
1=1 
  AND RSS.ID_RSS = (SELECT MAX(RSS2.ID_RSS)    FROM    TB_SYNTH_RSS RSS2    WHERE    RSS2.ID_SEJOUR = RSS.ID_SEJOUR) AND RSS.DATE_FIN >= '01/01/2026 00:00' AND RSS.DATE_FIN <= '31/12/2026 23:59'
