-- catalogue CORA : Analyse médico-administrative / Etude quantitative / etude quantitative des sejours / part de chaque type d'hospitalisation en nombre de sejours realisés
SELECT      UM.LIBELLE_UNITE_MED AS "LIBELLE UM RETENU" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RETENU" , 
    CASE WHEN UMU.HIST_TYPE_AUTORISATION = '01' THEN 'Réa'  WHEN RSS.CODE_CMD = '28' THEN 'Séance'  WHEN RSS.CODE_GHM LIKE '%J' THEN 'Ambu'  WHEN U.HJ = 'O' AND U.SEANCE <> 'O' AND U.AMBULATOIRE <> 'O' THEN 'HDJ'  WHEN U.TYPE_UNITE_PMSI = 'O' THEN 'Obstétrique' WHEN U.TYPE_UNITE_PMSI = 'C' THEN 'Chir' WHEN U.UHCD = 'C' THEN 'UHCD' ELSE 'Médecine' END AS "Type Hospit." , 
    RSS.ID_RSS AS "Nb de RSS" , 
    RSS.DUREE_SEJOUR AS "Durée" , 
    RSS.DATE_FIN AS "Date" , 
    NVL(RSS.MNT_TOTAL,0) AS "Montant" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUMX.ID_POLE, PRUMX.CODE_POLE, PRUMX.LIBELLE_POLE, PRUMX.ACTIF    FROM    TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUMX    WHERE    PUM2.ID_POLE=PRUMX.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )    AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU
     INNER JOIN TB_UNITE_M_UNITE UMU
      ON  UMU.ID_UNITE_MED = UM.ID_UNITE_MED AND ( UMU.DATE_DEBUT IS NULL OR UMU.DATE_DEBUT<=RSS.DATE_FIN ) AND ( UMU.DATE_FIN IS NULL OR UMU.DATE_FIN>=RSS.DATE_FIN )
     INNER JOIN TB_UNITE U
      ON  U.ID_UNITE = UMU.ID_UNITE

WHERE 
1=1 
  AND RSS.ID_RSS = (SELECT MAX(RSS2.ID_RSS)    FROM    TB_SYNTH_RSS RSS2    WHERE    RSS2.ID_SEJOUR = RSS.ID_SEJOUR) AND (UMU.DATE_DEBUT IS NULL OR UMU.DATE_DEBUT <= RSS.DATE_FIN) AND (UMU.DATE_FIN IS NULL OR UMU.DATE_FIN = RSS.DATE_FIN) AND RSS.DATE_FIN >= '01/01/2026 00:00' AND RSS.DATE_FIN <= '31/12/2026 23:59'
