-- catalogue CORA : Analyse médico-administrative / Etude quantitative / etude quantitative des sejours / sejours par domaine d'activité en nombre et en montant
SELECT      RSS.CODE_CMD || RSS.CODE_GHM AS "GHM" , 
    RSS.ID_RSS AS "RSS" , 
    RSS.CODE_CMD AS "CODE CMD" , 
    UM.LIBELLE_UNITE_MED AS "LIBELLE UM RETENU" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RETENU" , 
    SUBSTR(RSS.CODE_GHM,4,1) AS "SUFFIXE CODE GHM" , 
    PLGT.CODE_GHM_RACINE AS "RACINE GHM" , 
    RSS.DP_RETENU AS "DP RETENU" , 
    RSS.DATE_FIN AS "DATE" , 
    (RSS.DUREE_SEJOUR) AS "DUREES" , 
    (RSS.NB_JOUR_EXB) AS "NB JOURS EXB" , 
    (RSS.MONTANT_GHS) AS "MNT GHS" , 
    (RSS.MONTANT_DEP) AS "MNT DEP" , 
    (RSS.MONTANT_REDU) AS "MNT REDU" , 
    (NVL(RSS.MNT_TOTAL,0)) AS "MNT TOTAL" , 
    (RSS.QUANTITE_J_DEP) AS "NB JOURS DEP" , 
    LDA.CODE_PMSI_REGROUP AS "Code DA" , 
    LDA.LIB_PMSI_REGROUP AS "Libellé DA" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN PMSI_LIB_GHS_TARIF PLGT
      ON  RSS.CODE_CMD=PLGT.CMD AND RSS.CODE_GHS=PLGT.CODE_GHS AND RSS.CODE_GHM=PLGT.CODE_GHM AND        RSS.DATE_FIN >= PLGT.DATE_DEBUT AND RSS.DATE_FIN<=PLGT.DATE_FIN
     LEFT JOIN PMSI_LIB_REGROUP LDA
      ON  PLGT.CODE_PMSI_REGROUP_DA=LDA.CODE_PMSI_REGROUP AND LDA.TYPE_PMSI_REGROUP='DA'
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUM.ID_POLE, PRUM.CODE_POLE, PRUM.LIBELLE_POLE, PRUM.ACTIF    FROM    TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUM    WHERE    PUM2.ID_POLE=PRUM.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )    AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU

WHERE 
1=1 
  AND RSS.DATE_FIN >= '01/01/2026 00:00' AND RSS.DATE_FIN <= '31/12/2026 23:59'
