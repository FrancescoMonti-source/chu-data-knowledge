-- catalogue CORA : Analyse médico-administrative / liste patients
SELECT      RSS.ID_RSS AS "NB RSS" , 
    S.ID_SEJOUR AS "ID SEJOUR" , 
    UM.LIBELLE_UNITE_MED AS "LIBELLE UM RETENU" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RETENU" , 
    M.ID_MVT AS "ID MVT" , 
    P.NOM_USUEL_PATIENT AS "Nom Patient" , 
    P.PRENOM_PATIENT AS "Prénom Patient" , 
    S.NO_SEJOUR AS "N° Sejour" , 
    M.DATE_DEBUT_MVT AS "Date Déb Mvt" , 
    M.ID_UNITE_M AS "ID UNITE M" , 
    M.DATE_FIN_MVT AS "Date Fin Mvt" , 
    S.ID_MEDECIN_RESP_SEJ AS "ID MEDECIN RESP SEJ" , 
    M.ID_MEDECIN_RESPONSABLE AS "ID MEDECIN RESPONSABLE" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUMX.ID_POLE, PRUMX.CODE_POLE, PRUMX.LIBELLE_POLE, PRUMX.ACTIF    FROM    TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUMX    WHERE    PUM2.ID_POLE=PRUMX.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )    AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU,
    TB_SEJOUR S,
    TB_MVT M,
    TB_PATIENT P

WHERE 
1=1 
  AND P.ID_PATIENT=S.ID_PATIENT AND S.ID_SEJOUR=RSS.ID_SEJOUR AND RSS.ID_RSS=M.ID_RSS AND RSS.DATE_FIN>='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
