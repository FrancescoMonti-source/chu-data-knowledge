-- catalogue CORA : Analyse médico-administrative / Analyse activité / actes ccam par site
SELECT      UMVT.LIBELLE_UNITE AS "LIBELLE UNITE MVT" , 
    A.CODE_ACTE AS "CODE ACTE" , 
    NV.LIBELLE_LONG AS "LIBELLE ACTE" , 
    NVL(A.QUANTITE,0) AS "NB ACTES" , 
    UOA.nom_util || ' ' || UOA.prenom_util AS "MED EXEC ACTIV" , 
    RAIP.ICR AS "SOMME ICR" , 
    NVL(SADA.LIBELLE_SITE, 'Site N.R.') AS "LIBELLE SITE REAL ACTIV" , 
    M.DATE_FIN_MVT AS "DATE FIN MVT" 

FROM     TB_MVT M
     INNER JOIN TB_UNITE UMVT
      ON  M.ID_UNITE_M=UMVT.CODE_UNITE
     INNER JOIN TB_ACTE A
      ON  M.ID_MVT=A.ID_MVT
     INNER JOIN TB_DET_ACTE DA
      ON  A.ID_ACTE=DA.ID_ACTE
     LEFT JOIN UTIL UOA
      ON  DA.ID_MEDECIN=UOA.ID_UTIL
     LEFT JOIN TB_SITE SADA
      ON  DA.ID_SITE_ACTIVITE=SADA.ID_SITE
     INNER JOIN CCAM_NV NV
      ON  A.CODE_ACTE=NV.CLE
     INNER JOIN R_ACTE RA
      ON  NV.CLE=RA.COD_ACTE AND NV.DT_MODIF=RA.DT_MODIF
     INNER JOIN R_ACTE_IVITE RAI
      ON  RA.COD_ACTE=RAI.ACTE_COD AND RA.DT_MODIF=RAI.DT_MODIF AND DA.ID_ACTIVITE=RAI.ACTIV_COD
     INNER JOIN R_ACTE_IVITE_PHASE RAIP
      ON  RAI.COD_AA=RAIP.AA_COD AND RAI.DT_MODIF=RAIP.DT_MODIF AND A.PHASE_ACTE=RAIP.PHASE_COD

WHERE 
1=1 
  AND A.DATE_ACTE >= '01/01/2026 00:00' AND A.DATE_ACTE <= '31/12/2026 23:59'
