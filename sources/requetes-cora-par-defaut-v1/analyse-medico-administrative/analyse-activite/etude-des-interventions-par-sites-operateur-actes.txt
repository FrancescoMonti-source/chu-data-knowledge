-- catalogue CORA : Analyse médico-administrative / Analyse activité / etude des interventions par sites-operateur-actes
SELECT      M.ID_UNITE_M AS "CODE UNITE MVT" , 
    UMVT.LIBELLE_UNITE AS "LIBELLE UNITE MVT" , 
    P.SEXE AS "SEXE" , 
    (I.FIN_INTERVENTION - I.DEBUT_INTERVENTION) *1440 AS "DUREE INTERVENTION" , 
    count(distinct I.ID_INTERVENTION) AS "NB INTERV" , 
    A.CODE_ACTE AS "CODE ACTE" , 
    NV.LIBELLE_LONG AS "LIBELLE ACTE" , 
    A.DATE_ACTE AS "DATE ACTE" , 
    VueOp1.Nom AS "BLOC - OPER 1" 

FROM     TB_PATIENT P
     INNER JOIN TB_SEJOUR S
         INNER JOIN TB_MVT M
             INNER JOIN TB_INTERVENTION I
                 INNER JOIN TB_ACTE A
                     INNER JOIN CCAM_NV NV
                      ON  A.CODE_ACTE=NV.CLE
                  ON  I.ID_INTERVENTION=A.NO_INTERVENTION
                 LEFT JOIN (SELECT OP.ID_INTERVENTION, U.NOM_UTIL || ' ' || U.PRENOM_UTIL AS NOM   FROM TB_INTERVENTION_OPERATEUR OP, UTIL U  WHERE U.id_util=OP.ID_OPERATEUR AND OP.TYPE_INTERV='1') VueOp1
                  ON  I.ID_INTERVENTION=VueOp1.ID_INTERVENTION
              ON  M.ID_MVT=I.ID_MVT
             INNER JOIN TB_UNITE UMVT
              ON  M.ID_UNITE_PMSI=UMVT.ID_UNITE
          ON  S.ID_SEJOUR=M.ID_SEJOUR
      ON  P.ID_PATIENT=S.ID_PATIENT

WHERE 
1=1 
 AND A.DATE_ACTE >= '01/01/2026 00:00' AND A.DATE_ACTE <= '31/12/2026 23:59'

 GROUP BY M.ID_UNITE_M, UMVT.LIBELLE_UNITE, P.SEXE, (I.FIN_INTERVENTION - I.DEBUT_INTERVENTION) *1440, A.CODE_ACTE, NV.LIBELLE_LONG, A.DATE_ACTE, VueOp1.Nom
