-- catalogue CORA : Analyse médico-administrative / Analyse activité / actes-ivités par code ANP
SELECT      U.CODE_UNITE AS "Code UF" , 
    U.LIBELLE_UNITE AS "Libellé Unité" , 
    1 AS "Nb Activité" , 
    A.CODE_ACTE AS "Code Acte" , 
    DA.ID_ACTIVITE AS "Activité" , 
    DA.MNT_ACTIVITE AS "Prix Unitaire" , 
    CASE WHEN (NVL(DA.CODE_ASSOCIATION,'na')='na' OR DA.CODE_ASSOCIATION='1') AND DA.FACTURABLE='O' THEN '1 - 100%'       WHEN DA.CODE_ASSOCIATION='2' AND DA.FACTURABLE='O' THEN '2 - 50%'       WHEN DA.CODE_ASSOCIATION='3' AND DA.FACTURABLE='O' THEN '3 - 75%'       WHEN DA.CODE_ASSOCIATION='4' AND DA.FACTURABLE='O' THEN '4 - 100%'       WHEN DA.CODE_ASSOCIATION='4' AND DA.FACTURABLE='O' THEN '5 - 100%'       WHEN DA.FACTURABLE='N' OR DA.FACTURABLE IS NULL THEN 'Non Facturable'       ELSE 'Inconnu' END AS "Code ANP" 

FROM     TB_MVT M,
    TB_ACTE A,
    TB_DET_ACTE DA,
    TB_UNITE U

WHERE 
1=1 
  AND M.ID_MVT=A.ID_MVT AND M.ID_UNITE_M=U.CODE_UNITE AND A.ID_ACTE=DA.ID_ACTE AND A.DATE_ACTE >='01/01/2026 00:00' AND A.DATE_ACTE <= '31/12/2026 23:59'
 ORDER BY CASE WHEN (NVL(DA.CODE_ASSOCIATION,'na')='na' OR DA.CODE_ASSOCIATION='1') AND DA.FACTURABLE='O' THEN '1 - 100%'  WHEN DA.CODE_ASSOCIATION='2' AND DA.FACTURABLE='O' THEN '2 - 50%'  WHEN DA.CODE_ASSOCIATION='3' AND DA.FACTURABLE='O' THEN '3 - 75%'  WHEN DA.CODE_ASSOCIATION='4' AND DA.FACTURABLE='O' THEN '4 - 100%'  WHEN DA.CODE_ASSOCIATION='4' AND DA.FACTURABLE='O' THEN '5 - 100%'  WHEN DA.FACTURABLE='N' OR DA.FACTURABLE IS NULL THEN 'Non Facturable'  ELSE 'Inconnu' END ASC
