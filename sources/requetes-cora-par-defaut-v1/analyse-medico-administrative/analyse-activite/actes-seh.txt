-- catalogue CORA : Analyse médico-administrative / Analyse activité / actes SEH
SELECT      S.NO_SEJOUR AS "NUM SEJOUR" , 
    floor(S.DATE_SORTIE-S.DATE_ENTREE)+1 AS "DUREE SEJ" , 
    M.ID_UNITE_M AS "CODE UNITE MVT" , 
    UMVT.LIBELLE_UNITE AS "LIBELLE UNITE MVT" , 
    A.DATE_ACTE AS "DATE ACTE" , 
    A.CODE_ACTE AS "CODE ACTE" , 
    NV.LIBELLE_LONG AS "LIBELLE ACTE" , 
    SEA.LIBELLE_SITE AS "LIBELLE SITE EXEC ACTE" , 
    NV.LISTE_ANNEXE_9 AS "ANNEXE_9" , 
    UOA.nom_util || ' ' || UOA.prenom_util AS "MED EXEC ACTIV" , 
    sum(NVL(A.QUANTITE,0)) AS "NB ACTES" , 
    M.BLOCAGE_RSS AS "RSS EXCLU" , 
    UMVT.TYPEU AS "TYPE UNITE MVT" 

FROM     TB_SEJOUR S      INNER JOIN TB_MVT M          INNER JOIN TB_ACTE A              INNER JOIN TB_DET_ACTE DA                  LEFT JOIN Util UOA                   ON  DA.ID_MEDECIN=UOA.id_util               ON  A.ID_ACTE=DA.ID_ACTE              INNER JOIN CCAM_NV NV               ON  A.CODE_ACTE=NV.CLE              INNER JOIN TB_SITE SEA               ON  A.ID_SITE=SEA.ID_SITE              LEFT JOIN TB_COTATION_NGAP N               ON  A.NO_INTERVENTION_CCAM=N.NO_INTERVENTION_CCAM AND N.COTATION LIKE 'SE%'           ON  M.ID_MVT=A.ID_MVT          INNER JOIN (SELECT ID_UNITE, LIBELLE_UNITE, CASE WHEN GESTION_PMSI = 'O' THEN 'MCO' WHEN EXTERNE = 'O' THEN 'EXT' WHEN URGENCE = 'O' THEN 'URG' END  AS  TYPEU  FROM  TB_UNITE) UMVT           ON  M.ID_UNITE_PMSI=UMVT.ID_UNITE       ON  S.ID_SEJOUR=M.ID_SEJOUR 

WHERE 
1=1 
  AND A.DATE_ACTE >= '01/01/2026 00:00' AND A.DATE_ACTE <= '31/12/2026 23:59' AND S.DATE_ENTREE = S.DATE_SORTIE AND NV.LISTE_ANNEXE_9 IN ('1','2','3','4','5','6') AND N.CPT_ID_COTATION_NGAP IS NULL

 GROUP BY S.NO_SEJOUR, floor(S.DATE_SORTIE-S.DATE_ENTREE)+1, M.ID_UNITE_M, UMVT.LIBELLE_UNITE, A.DATE_ACTE, A.CODE_ACTE, NV.LIBELLE_LONG, SEA.LIBELLE_SITE, NV.LISTE_ANNEXE_9, UOA.nom_util || ' ' || UOA.prenom_util, M.BLOCAGE_RSS, UMVT.TYPEU
