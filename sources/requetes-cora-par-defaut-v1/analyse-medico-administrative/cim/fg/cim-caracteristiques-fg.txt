-- catalogue CORA : Analyse médico-administrative / CIM / FG / CIM - caracteristiques FG
SELECT      FG.CARACTERISTIQUE AS "Code" , 
    CARD.LIBELLE AS "Libellé" , 
    COUNT(*) AS "Nb Diags" , 
    AVG(RSS.DUREE_SEJOUR) AS "DMS" 

FROM     TB_SYNTH_RSS RSS,
    TB_MVT M,
    TB_DIAG D,
    CORA_CCAM.FG_DIAG_10CR FG,
    CORA_CCAM.FG_LIBCARD CARD

WHERE 
1=1 
  AND RSS.ID_RSS=M.ID_RSS AND M.ID_MVT=D.ID_MVT AND D.CODE_DIAG=FG.CODE_DIAG AND RTRIM(FG.CARACTERISTIQUE) = RTRIM(CARD.CARACTERISTIQUE) AND RSS.DATE_FIN>= '01/01/2026 00:00' AND RSS.DATE_FIN<= '31/12/2026 23:59' AND FG.VERSION_FG = (SELECT max(VERSION_FG) FROM FG_LIBCARD) AND FG.VERSION_FG =CARD.VERSION_FG AND (CARD.CODE_CMD IS NULL OR CARD.CODE_CMD='XX' OR CARD.CODE_CMD=RSS.CODE_CMD)

 GROUP BY FG.CARACTERISTIQUE, CARD.LIBELLE
 ORDER BY CARD.LIBELLE ASC
