-- catalogue CORA : Analyse médico-administrative / CIM / FG / CIM - caracteristiques FG - Diags
SELECT      D.CODE_DIAG AS "Code Diag" , 
    CIM.LIB AS "Libellé" , 
    COUNT(*) AS "Nb Diags" , 
    AVG(RSS.DUREE_SEJOUR) AS "DMS" 

FROM     TB_SYNTH_RSS RSS,
    TB_MVT M,
    TB_DIAG D,
    CORA_CCAM.FG_DIAG_10CR FG,
    CORA_CCAM.CIM_HIERARCHIE CIM

WHERE 
1=1 
  AND RSS.ID_RSS=M.ID_RSS AND M.ID_MVT=D.ID_MVT AND D.CODE_DIAG=FG.CODE_DIAG AND D.CODE_DIAG=CIM.CODEFILS AND RSS.DATE_FIN>= '01/01/2026 00:00' AND RSS.DATE_FIN<= '31/12/2026 23:59' AND FG.VERSION_FG= (SELECT MAX(VERSION_FG) FROM FG_LIBCARD)

 GROUP BY D.CODE_DIAG, CIM.LIB
 ORDER BY D.CODE_DIAG ASC, CIM.LIB ASC
