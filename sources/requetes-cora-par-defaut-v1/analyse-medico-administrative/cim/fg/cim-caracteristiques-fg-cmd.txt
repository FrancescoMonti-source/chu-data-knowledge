-- catalogue CORA : Analyse médico-administrative / CIM / FG / CIM - caracteristiques FG - CMD
SELECT      RSS.CODE_CMD AS "Code CMD" , 
    RSS.CODE_CMD || RSS.CODE_GHM AS "Code GHM" , 
    PLGT.LIBELLE_GHS AS "Libellé GHM" , 
    COUNT(*) AS "Nb Diags" , 
    AVG(RSS.DUREE_SEJOUR) AS "DMS" 

FROM     TB_SYNTH_RSS RSS,
    PMSI_LIB_GHS_TARIF PLGT,
    TB_MVT M,
    TB_DIAG D

WHERE 
1=1 
  AND RSS.ID_RSS=M.ID_RSS AND PLGT.CODE_GHM=RSS.CODE_GHM AND PLGT.CMD=RSS.CODE_CMD AND PLGT.DATE_DEBUT<=RSS.DATE_FIN AND PLGT.DATE_FIN>=RSS.DATE_FIN AND M.ID_MVT=D.ID_MVT AND RSS.DATE_FIN>= '01/01/2026 00:00' AND RSS.DATE_FIN<= '31/12/2026 23:59'

 GROUP BY RSS.CODE_CMD, RSS.CODE_CMD || RSS.CODE_GHM, PLGT.LIBELLE_GHS
 ORDER BY RSS.CODE_CMD ASC, RSS.CODE_CMD || RSS.CODE_GHM ASC, PLGT.LIBELLE_GHS ASC
