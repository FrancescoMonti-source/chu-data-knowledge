-- catalogue CORA : Analyse médico-administrative / CIM / Hiérarchie / CIM - Hiérarchie - DP
SELECT      VueChap.LIB AS "Libellé Chapitre" , 
    VueSousChap.LIB AS "Libellé Sous-Chapitre" , 
    VueCategorie.LIB AS "Libellé Catégorie" , 
    RSS.DP_RETENU AS "DP Retenu" , 
    CIM.LIB AS "Libellé DP Retenu" , 
    RSS.ID_RSS AS "NB RSS" , 
    DUREE_SEJOUR AS "DMS" , 
    UM.CODE_UNITE_MED AS "Code UM Retenue" , 
    UM.LIBELLE_UNITE_MED AS "Libellé UM Retenue" , 
    PRUM.CODE_POLE AS "Code Pôle Retenu" , 
    PRUM.LIBELLE_POLE AS "Libellé Pôle Retenu" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUM.ID_POLE, PRUM.CODE_POLE, PRUM.LIBELLE_POLE, PRUM.ACTIF    FROM    TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUM    WHERE    PUM2.ID_POLE=PRUM.ID_POLE) PRUM
      ON  (PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )   AND (PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU
     INNER JOIN CORA_CCAM.CIM_HIERARCHIE CIM
      ON  RSS.DP_RETENU=CIM.CODEFILS
     LEFT JOIN (SELECT HIERARCHIE, LIB, CHAP, SSCHAP, CAT    FROM    CORA_CCAM.CIM_HIERARCHIE    WHERE    SSCHAP='00' AND CAT='00' AND SSCAT='00') VUECHAP
      ON  CIM.CHAP=VUECHAP.CHAP
     LEFT JOIN (SELECT HIERARCHIE, LIB, CHAP, SSCHAP, CAT    FROM    CORA_CCAM.CIM_HIERARCHIE    WHERE    SSCHAP!='00' AND CAT='00' AND SSCAT='00') VUESOUSCHAP
      ON  CIM.CHAP=VUESOUSCHAP.CHAP AND CIM.SSCHAP=VUESOUSCHAP.SSCHAP
     LEFT JOIN (SELECT HIERARCHIE, LIB, CHAP, SSCHAP, CAT    FROM    CORA_CCAM.CIM_HIERARCHIE    WHERE    SSCHAP!='00' AND CAT!='00' AND SSCAT='00') VUECATEGORIE
      ON  CIM.CHAP=VUECATEGORIE.CHAP AND CIM.SSCHAP=VUECATEGORIE.SSCHAP AND CIM.CAT=VUECATEGORIE.CAT

WHERE 
1=1 
  AND RSS.DATE_FIN>='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
 ORDER BY VueCategorie.CHAP ASC, VueCategorie.SSCHAP ASC, VueCategorie.CAT ASC, RSS.DP_RETENU ASC
