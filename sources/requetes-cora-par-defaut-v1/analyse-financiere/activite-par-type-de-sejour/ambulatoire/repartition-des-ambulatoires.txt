-- catalogue CORA : Analyse Financière / Activité par Type de Séjour / Ambulatoire / repartition des ambulatoires
SELECT      RSS.CODE_CMD || RSS.CODE_GHM AS "GHM RSS" , 
    S.DATE_SORTIE AS "DATE SORTIE SEJ" , 
    M.MODE_ENTREE AS "MODE ENTREE MVT" , 
    floor((M.DATE_DEBUT_MVT - P.DATE_NAIS)/365.25) AS "AGE PATIENT ANNEES ENT MVT" , 
    M.ID_UNITE_M AS "CODE UNITE MVT" , 
    UMVT.LIBELLE_UNITE AS "LIBELLE UNITE MVT" , 
    A.CODE_ACTE AS "CODE ACTE" , 
    NV.LIBELLE_LONG AS "LIBELLE ACTE" , 
    RSS.ID_RSS AS "ID_RSS" 

FROM     TB_PATIENT P
     INNER JOIN TB_SEJOUR S
         INNER JOIN TB_SYNTH_RSS RSS
             INNER JOIN TB_SYNTH_RUM RUM
                 INNER JOIN TB_MVT_RUM MVTRUM
                     INNER JOIN TB_MVT M
                         INNER JOIN TB_ACTE A
                             INNER JOIN CCAM_NV NV
                              ON  A.CODE_ACTE=NV.CLE
                          ON  M.ID_MVT=A.ID_MVT
                         INNER JOIN TB_UNITE UMVT
                          ON  M.ID_UNITE_PMSI=UMVT.ID_UNITE
                      ON  MVTRUM.ID_MVT=M.ID_MVT
                  ON  RUM.ID_RUM=MVTRUM.ID_RUM
              ON  RSS.ID_RSS=RUM.ID_RSS
          ON  S.ID_SEJOUR=RSS.ID_SEJOUR
      ON  P.ID_PATIENT=S.ID_PATIENT

WHERE 
1=1 
 AND RSS.DATE_FIN >= '01/01/2026 00:00' AND RSS.DATE_FIN <= '31/12/2026 23:59' AND RSS.CODE_CMD || RSS.CODE_GHM LIKE '%' || 'J'
