-- catalogue CORA : Analyse Financière / Activité par Type de Séjour / Ambulatoire / Gradation / Ambu éligibles FG gradation, variables non recueillies
SELECT      S.NO_SEJOUR AS "NUM SEJOUR" , 
    RSS.CODE_CMD || RSS.CODE_GHM AS "GHM RSS" , 
    RSS.DATE_DEBUT AS "DATE DEBUT RSS" , 
    floor(S.DATE_SORTIE-S.DATE_ENTREE) AS "DUREE SEJ" , 
    UMRSS.CODE_UNITE_MED AS "CODE UNITE RETENUE RSS" , 
    RSS.ELIGIBLE_GRADATION_HDJ AS "RSS ELIGIBLE GRADATION" , 
    RSS.HDJ_RELEVE_EXT AS "RSS RELEVE EXT" , 
    RSS.CODE_ERR AS "CODE ERR RSS" 

FROM     TB_SEJOUR S
     INNER JOIN TB_SYNTH_RSS RSS
         INNER JOIN TB_SYNTH_RUM RUM
             INNER JOIN TB_MVT_RUM MVTRUM
                 INNER JOIN TB_MVT M
                     LEFT JOIN TB_INSTRUCTION_GRADATION GRAD
                      ON  M.ID_MVT = GRAD.ID_MVT
                  ON  MVTRUM.ID_MVT=M.ID_MVT
              ON  RUM.ID_RUM=MVTRUM.ID_RUM
          ON  RSS.ID_RSS=RUM.ID_RSS
         INNER JOIN TB_UNITE_MEDICALE UMRSS
          ON  RSS.ID_UM_RETENU=UMRSS.ID_UNITE_MED
      ON  S.ID_SEJOUR=RSS.ID_SEJOUR

WHERE 
1=1 
  AND RSS.ELIGIBLE_GRADATION_HDJ = '1' AND GRAD.CAT_NB_INTERVENTIONS_TOTALES IS NULL AND GRAD.CTXT_PAT_SURV_PART IS NULL AND GRAD.ADMIN_PRODUIT_RH IS NULL AND RSS.DATE_DEBUT >= '01/03/2020 00:00' AND RSS.DATE_DEBUT <= '31/12/2020 23:59'
