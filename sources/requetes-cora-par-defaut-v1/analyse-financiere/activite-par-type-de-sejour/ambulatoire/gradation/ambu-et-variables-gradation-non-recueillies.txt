-- catalogue CORA : Analyse Financière / Activité par Type de Séjour / Ambulatoire / Gradation / Ambu et variables gradation non recueillies
SELECT      S.NO_SEJOUR AS "NUM SEJOUR" , 
    S.DATE_ENTREE AS "DATE ENTREE SEJ" , 
    S.UF_ENTREE AS "UF ENTREE SEJ" , 
    S.MODE_ENTREE AS "MODE ENTREE SEJ" , 
    S.CODE_PROVENANCE AS "CODE PROV SEJ" 

FROM     TB_SEJOUR S
     INNER JOIN TB_MVT M
         LEFT JOIN TB_INSTRUCTION_GRADATION GRAD
          ON  M.ID_MVT = GRAD.ID_MVT
      ON  S.ID_SEJOUR=M.ID_SEJOUR
     INNER JOIN TB_UNITE U
      ON  M.ID_UNITE_PMSI=U.ID_UNITE

WHERE 
1=1 
  AND S.DATE_SORTIE >= '01/03/2020 00:00' AND S.DATE_SORTIE <= '31/12/2020 23:59' AND floor(S.DATE_SORTIE-S.DATE_ENTREE) = 0 AND GRAD.CTXT_PAT_SURV_PART IS NULL AND GRAD.CAT_NB_INTERVENTIONS_TOTALES IS NULL AND GRAD.ADMIN_PRODUIT_RH IS NULL AND U.GESTION_PMSI='O'
