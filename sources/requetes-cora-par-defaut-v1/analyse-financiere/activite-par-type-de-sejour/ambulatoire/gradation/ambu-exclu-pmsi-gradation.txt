-- catalogue CORA : Analyse Financière / Activité par Type de Séjour / Ambulatoire / Gradation / Ambu exclu PMSI (gradation)
SELECT      S.NO_SEJOUR AS "NUM SEJOUR" , 
    S.DATE_ENTREE AS "DATE ENTREE SEJ" , 
    NVL(DA.MNT_TOTAL,0) AS "MONTANT TOTAL" , 
    NVL(NGAPLC.MNT_LC, 0) + NVL(NGAP.MNT_DEPASS, 0) AS "MONTANT FACTURE NGAP" 

FROM     TB_SEJOUR S
     INNER JOIN TB_MVT M
         LEFT JOIN TB_ACTE A
             INNER JOIN TB_DET_ACTE DA
              ON  A.ID_ACTE=DA.ID_ACTE
          ON  M.ID_MVT=A.ID_MVT
         LEFT JOIN TB_COTATION_NGAP NGAP
             INNER JOIN TB_NGAP_LC NGAPLC
              ON  NGAP.CPT_ID_COTATION_NGAP=NGAPLC.CPT_ID_COTATION_NGAP
          ON  M.ID_MVT=NGAP.ID_MVT
         LEFT JOIN TB_INSTRUCTION_GRADATION GRAD
          ON  M.ID_MVT = GRAD.ID_MVT
      ON  S.ID_SEJOUR=M.ID_SEJOUR

WHERE 
1=1 
  AND S.DATE_ENTREE >= '01/03/2020 00:00' AND S.DATE_ENTREE <= '31/12/2020 23:59' AND floor(S.DATE_SORTIE-S.DATE_ENTREE) = 0 AND M.BLOCAGE_RSS = 'O' AND GRAD.CAT_NB_INTERVENTIONS_TOTALES = 'A'
