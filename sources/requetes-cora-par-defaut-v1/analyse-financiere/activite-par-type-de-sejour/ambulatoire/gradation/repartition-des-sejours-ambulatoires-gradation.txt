-- catalogue CORA : Analyse Financière / Activité par Type de Séjour / Ambulatoire / Gradation / repartition des sejours ambulatoires (gradation)
SELECT      RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    CASE WHEN RSS.DATE_DEBUT = RSS.DATE_FIN THEN 0 ELSE RSS.DUREE_SEJOUR END AS "DUREE REELLE RSS" , 
    RSS.DATE_DEBUT AS "DATE DEBUT RSS" , 
    RSS.ELIGIBLE_GRADATION_HDJ AS "RSS ELIGIBLE GRADATION" , 
    RSS.HDJ_RELEVE_EXT AS "RSS RELEVE EXT" , 
    RSS.RESCRIT_TARIFAIRE AS "RSS RESCRIT TARIFAIRE" , 
    CASE GRAD.CAT_NB_INTERVENTIONS_TOTALES WHEN 'A' THEN 'A - Inférieure à 3 interventions' WHEN 'B' THEN 'B - Egale à 3 interventions' WHEN 'C' THEN 'C - Supérieure à 3 interventions' ELSE 'Non renseigné' END AS "MVT - CAT NB INTERVENTIONS" , 
    CASE GRAD.ADMIN_PRODUIT_RH WHEN '2' THEN '2 - Non' WHEN '1' THEN '1 - Oui' ELSE 'Non renseigné' END AS "MVT - ADMIN PRODUIT RH" , 
    CASE GRAD.CTXT_PAT_SURV_PART WHEN '2' THEN '2 - Non' WHEN '1' THEN '1 - Oui' ELSE 'Non renseigné' END AS "MVT - CTXT PATIENT" , 
    SUM(nvl(RSS.MONTANT_GHS,0)+nvl(RSS.MONTANT_DEP,0)-nvl(RSS.MONTANT_COEF,0)+nvl(RSS.MONTANT_URG,0)+nvl(RSS.MONTANT_REA,0)-nvl(RSS.MONTANT_REDU,0)+nvl(RSS.MT_24Z05Z,0)+nvl(RSS.MT_24Z06Z,0)+nvl(RSS.MT_24Z07Z,0)+nvl(RSS.MT_DIALYSE,0)+nvl(RSS.MNT_JOUR_NEO1,0)+nvl(RSS.MNT_JOUR_NEO2,0)+nvl(RSS.MNT_JOUR_NEO3,0)+nvl(RSS.MNT_JOUR_SOINS_INT,0)+nvl(RSS.MNT_JOUR_SURV_CONT,0)+nvl(RSS.MNT_JOUR_SOINS_PALL,0)+nvl(RSS.MNT_JOUR_SOINS_INT_REA,0)+nvl(RSS.MNT_PRELEV_ORGANE,0)+nvl(RSS.MNT_SRA,0)+nvl(RSS.MNT_REP,0)+nvl(RSS.MNT_CAISSON_HB,0)+nvl(RSS.MNT_APHERESE,0)+nvl(RSS.MNT_DIP,0)+nvl(RSS.MNT_RDTH,0)+nvl(RSS.MNT_ANTEPARTUM,0)+nvl(RSS.MNT_RDTH_PED,0)) AS "SOMME MNT TOTAL RSS" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN TB_SYNTH_RUM RUM
         INNER JOIN TB_MVT_RUM MVTRUM
             INNER JOIN TB_MVT M
                 LEFT JOIN TB_INSTRUCTION_GRADATION GRAD
                  ON  M.ID_MVT = GRAD.ID_MVT
              ON  MVTRUM.ID_MVT=M.ID_MVT
          ON  RUM.ID_RUM=MVTRUM.ID_RUM
      ON  RSS.ID_RSS=RUM.ID_RSS

WHERE 
1=1 
 AND RSS.DATE_FIN >= '01/03/2020 00:00' AND RSS.DATE_FIN <= '31/12/2020 23:59' AND CASE WHEN RSS.DATE_DEBUT = RSS.DATE_FIN THEN 0 ELSE RSS.DUREE_SEJOUR END = 0

 GROUP BY RSS.NO_SEJOUR, CASE WHEN RSS.DATE_DEBUT = RSS.DATE_FIN THEN 0 ELSE RSS.DUREE_SEJOUR END, RSS.DATE_DEBUT, RSS.ELIGIBLE_GRADATION_HDJ, RSS.HDJ_RELEVE_EXT, RSS.RESCRIT_TARIFAIRE, CASE GRAD.CAT_NB_INTERVENTIONS_TOTALES WHEN 'A' THEN 'A - Inférieure à 3 interventions' WHEN 'B' THEN 'B - Egale à 3 interventions' WHEN 'C' THEN 'C - Supérieure à 3 interventions' ELSE 'Non renseigné' END, CASE GRAD.ADMIN_PRODUIT_RH WHEN '2' THEN '2 - Non' WHEN '1' THEN '1 - Oui' ELSE 'Non renseigné' END, CASE GRAD.CTXT_PAT_SURV_PART WHEN '2' THEN '2 - Non' WHEN '1' THEN '1 - Oui' ELSE 'Non renseigné' END
