-- catalogue CORA : Analyse Financière / Activité par Type de Séjour / Séances / analyse seances pendant hospitalisation
SELECT      NVL(RSS.DP_RETENU, 'Code Diag N.R.') AS "DP RETENU" , 
    NVL(UM.LIBELLE_UNITE_MED, 'Libellé UM N.R.') AS "LIBELLE UM RETENU" , 
    NVL(UM.CODE_UNITE_MED, 'Code UM N.R.') AS "CODE UM RETENU" , 
    NVL(PRUM.LIBELLE_POLE, 'Libellé Pôle N.R.') AS "LIBELLE POLE RETENU" , 
    NVL(PRUM.CODE_POLE, 'Code Pôle N.R.') AS "CODE POLE RETENU" , 
    NVL(CIM.LIB, 'Libellé Diag N.R.') AS "LIBELLE DIAG" , 
    RSS.ID_RSS AS "NB RSS" , 
    RSS.CODE_CMD AS "CODE CMD RSS" , 
    RSS.CODE_CMD || RSS.CODE_GHM AS "CODE GHM RSS" , 
    PLGT.CODE_GHM_RACINE AS "CODE RACINE GHM" , 
    PLGT.LIBELLE_GHS AS "LIBELLE GHM" , 
    RSS.DATE_FIN AS "DATE FIN" , 
    RSS.DUREE_SEJOUR AS "DUREE" , 
    RSS.NB_DIALYSE AS "NB DIALYSE" , 
    RSS.NB_CAISSON_HB AS "NB CAISSON HYPERBARE" , 
    RSS.NB_APHERESE AS "NB APHERESES" , 
    RSS.NB_DIP AS "NB DIP" , 
    RSS.NB_RDTH AS "NB RDTH" , 
    RSS.NB_RDTH_PED AS "NB RDTH PED" , 
    RSS.MT_DIALYSE AS "MONTANT DIALYSE" , 
    RSS.MNT_CAISSON_HB AS "MONTANT CAISSON HYPERBARE" , 
    RSS.MNT_APHERESE AS "MONTANT APHERESE" , 
    RSS.MNT_RDTH AS "MONTANT RDTH" , 
    RSS.MNT_DIP AS "MONTANT DIP" , 
    RSS.MNT_RDTH_PED AS "MONTANT RDTH PED" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN PMSI_LIB_GHS_TARIF PLGT
      ON  RSS.CODE_CMD=PLGT.CMD AND RSS.CODE_GHS=PLGT.CODE_GHS AND RSS.CODE_GHM=PLGT.CODE_GHM AND  RSS.DATE_FIN >= PLGT.DATE_DEBUT AND RSS.DATE_FIN<=PLGT.DATE_FIN
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED,        PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUMX.ID_POLE, PRUMX.CODE_POLE, PRUMX.LIBELLE_POLE, PRUMX.ACTIF     FROM     TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUMX     WHERE     PUM2.ID_POLE=PRUMX.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN ) AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UM
      ON  UM.ID_UNITE_MED = RSS.ID_UM_RETENU
     LEFT JOIN CIM_HIERARCHIE CIM
      ON  RSS.DP_RETENU=CIM.CODEFILS

WHERE 
1=1 
  AND (RSS.NB_DIALYSE>1 OR RSS.NB_CAISSON_HB>1 OR RSS.NB_APHERESE>1 OR RSS.NB_RDTH>1 OR RSS.NB_RDTH_PED>1) AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <= '31/12/2026 23:59'
