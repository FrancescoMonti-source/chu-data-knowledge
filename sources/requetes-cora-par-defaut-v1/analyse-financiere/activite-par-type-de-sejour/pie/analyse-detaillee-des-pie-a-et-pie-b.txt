-- catalogue CORA : Analyse Financière / Activité par Type de Séjour / PIE / analyse detaille des PIE A et PIE B
SELECT      RSS.ID_RSS AS "NB RSS" , 
    RSS.DUREE_SEJOUR AS "DUREE RSS" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    D.CODE_DIAG AS "CODE DIAG" , 
    D.TYPE_DIAG AS "TYPE DIAG" , 
    RSS.MODE_ENTREE AS "MODE ENTREE RSS" , 
    RSS.MODE_SORTIE AS "MODE SORTIE RSS" , 
    CASE WHEN D.CODE_DIAG='Z7580'  THEN 'PIE A' ELSE 'PIE B' END AS "PIE" 

FROM     TB_SYNTH_RSS RSS,
    TB_SYNTH_RUM RUM,
    TB_MVT_RUM MVTRUM,
    TB_MVT M,
    TB_DIAG D

WHERE 
1=1 
  AND RSS.ID_RSS=RUM.ID_RSS AND RUM.ID_RUM=MVTRUM.ID_RUM AND MVTRUM.ID_MVT=M.ID_MVT AND M.ID_MVT=D.ID_MVT AND (D.CODE_DIAG='Z7580' OR (RSS.MODE_ENTREE='0' AND RSS.MODE_SORTIE='0')) AND RSS.DATE_FIN>='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
