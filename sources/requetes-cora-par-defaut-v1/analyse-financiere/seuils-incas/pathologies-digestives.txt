SELECT      'Chirurgie Digestive' AS "Chirurgie Digestive" , 
    RSS.ID_RSS AS "NUM RSS" , 
    RSS.CODE_GHM AS "CODE GHM RSS" , 
    RSS.DP_RETENU AS "DP RETENU" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSS.AGE_ENTREE AS "AGE PATIENT ANNEES ENTREE RSS" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN TB_UNITE_MEDICALE UM
         LEFT JOIN TB_UNITE_M_UNITE UMU
             LEFT JOIN TB_EG_UNITE EGU
                 LEFT JOIN TB_EG EG
                  ON  EGU.ID_EG = EG.ID_EG
              ON  UMU.ID_UNITE = EGU.ID_UNITE
          ON  UM.ID_UNITE_MED = UMU.ID_UNITE_MED
      ON  RSS.ID_UM_RETENU = UM.ID_UNITE_MED

WHERE 
1=1 
  AND RSS.DP_RETENU  IN ('C150','C151','C152','C153','C154','C155','C158','C159','C160','C161','C162','C163','C164','C165','C166','C168','C169','C169||0','C169||8','C170','C171','C172','C173','C178','C179','C180','C181','C182','C183','C184','C185','C186','C187','C188','C189','C189||0','C189||8','C19','C20','C210','C211','C212','C218','C220','C221','C222','C223','C224','C227','C229','C23','C240','C241','C248','C249','C250','C251','C252','C253','C254','C254||0','C254||8','C257','C258','C259','C259||0','C259||8','C261','C268','C451','C467','C481','C482','C488','C784','C785','C786','C787','C788','D001','D002','D010','D011','D012','D013','D014','D015','D017','D371','D372','D373','D374','D375','D376','D377','D4830','D484') AND SUBSTR(RSS.CODE_GHM,1,1)='C' AND RSS.AGE_ENTREE >= '18' AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
