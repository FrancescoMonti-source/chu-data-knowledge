-- catalogue CORA : Analyse Financière / Seuils INCas / pathologies gyneco
SELECT      'Chirurgie Gynécologique' AS "Chirurgie Gynécologique" , 
    RSS.CODE_CMD AS "Code CMD" , 
    RSS.ID_RSS AS "NUM RSS" , 
    RSS.CODE_GHM AS "CODE GHM RSS" , 
    RSS.DP_RETENU AS "DP RETENU" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSS.AGE_ENTREE AS "AGE PATIENT ANNEES ENTREE RSS" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN TB_UNITE_MEDICALE UM
         LEFT JOIN TB_UNITE_M_UNITE UMU
             LEFT JOIN TB_EG_UNITE EGU
                 LEFT JOIN TB_EG EG
                  ON  EGU.ID_EG = EG.ID_EG
              ON  UMU.ID_UNITE = EGU.ID_UNITE
          ON  UM.ID_UNITE_MED = UMU.ID_UNITE_MED
      ON  RSS.ID_UM_RETENU = UM.ID_UNITE_MED

WHERE 
1=1 
  AND RSS.DP_RETENU  IN ('C510','C511','C512','C518','C519','C52','C530','C531','C538','C539','C540','C541','C542','C543','C548','C549','C55','C56','C570','C571','C572','C573','C574','C577','C578','C58 ','C796','D390','D391','D392','D397') AND SUBSTR(RSS.CODE_GHM,1,1)='C' AND RSS.CODE_CMD||RSS.CODE_GHM  NOT IN ('13C11J','13C111','13C112','13C113','13C114') AND RSS.AGE_ENTREE >= '18' AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
