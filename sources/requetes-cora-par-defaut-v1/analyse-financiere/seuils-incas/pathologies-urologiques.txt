-- catalogue CORA : Analyse Financière / Seuils INCas / pathologies urologiques
SELECT      'Chirurgie Urologique' AS "Chirurgie Urologique" , 
    RSS.ID_RSS AS "NUM RSS" , 
    RSS.CODE_CMD AS "Code CMD" , 
    RSS.CODE_GHM AS "CODE GHM RSS" , 
    RSS.DP_RETENU AS "DP RETENU" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSS.AGE_ENTREE AS "AGE PATIENT ANNEES ENTREE RSS" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN TB_UNITE_MEDICALE UM
         LEFT JOIN TB_UNITE_M_UNITE UMU
             LEFT JOIN TB_EG_UNITE EGU
                 LEFT JOIN TB_EG EG
                  ON  EGU.ID_EG = EG.ID_EG
              ON  UMU.ID_UNITE = EGU.ID_UNITE
          ON  UM.ID_UNITE_MED = UMU.ID_UNITE_MED
      ON  RSS.ID_UM_RETENU = UM.ID_UNITE_MED

WHERE 
1=1 
  AND RSS.DP_RETENU  IN ('C480','C600','C601','C602','C608','C609','C61','C620','C621','C629','C630','C631','C632','C637','C638','C64','C65','C66','C670','C671','C672','C673','C674','C675','C676','C677','C678','C679','C680','C681','C688','C740','C741','C749','C790','C791','C797','D074','D075','D090','D400','D401','D407','D409','D410','D411','D412','D413','D414','D417','D441','D483','D4838') AND SUBSTR(RSS.CODE_GHM,1,1)='C' AND RSS.CODE_CMD||RSS.CODE_GHM  NOT IN ('11C13J','11C131','11C132','11C133','11C134','12C041','12C042','12C043','12C043') AND RSS.AGE_ENTREE >= '18' AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
