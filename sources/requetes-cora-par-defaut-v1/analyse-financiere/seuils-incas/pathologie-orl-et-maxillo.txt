SELECT      'Chirurgie ORL et Maxillo-Faciale' AS "Chir ORL et Maxillo-Faciale" , 
    RSS.ID_RSS AS "NUM RSS" , 
    RSS.CODE_GHM AS "CODE GHM RSS" , 
    RSS.DP_RETENU AS "DP RETENU" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSS.AGE_ENTREE AS "AGE PATIENT ANNEES ENTREE RSS" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN TB_UNITE_MEDICALE UM
         LEFT JOIN TB_UNITE_M_UNITE UMU
             LEFT JOIN TB_EG_UNITE EGU
                 LEFT JOIN TB_EG EG
                  ON  EGU.ID_EG = EG.ID_EG
              ON  UMU.ID_UNITE = EGU.ID_UNITE
          ON  UM.ID_UNITE_MED = UMU.ID_UNITE_MED
      ON  RSS.ID_UM_RETENU = UM.ID_UNITE_MED

WHERE 
1=1 
  AND RSS.DP_RETENU  IN ('C000','C001','C002','C003','C004','C005','C006','C008','C009','C01','C020','C021','C022','C023','C024','C028','C029','C030','C031','C039','C040','C041','C048','C049','C050','C051','C052','C058','C059','C060','C061','C062','C068','C069','C07','C080','C081','C088','C089','C090','C091','C098','C099','C100','C101','C102','C103','C104','C108','C109','C110','C111','C112','C113','C118','C119','C12','C130','C131','C132','C138','C139','C140','C142','C148','C300','C301','C310','C311','C312','C313','C318','C319','C320','C321','C322','C323','C328','C329','C462','C4670','C770','D000','D020','D370','D380','C73 ','C750','D093','D440','D442') AND SUBSTR(RSS.CODE_GHM,1,1)='C' AND RSS.AGE_ENTREE >= '18' AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
