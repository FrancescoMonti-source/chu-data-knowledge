-- catalogue CORA : Analyse Financière / Seuils INCas / pathologie thoraciques
SELECT      'Chirurgie Thoracique' AS "Chirurgie Thoracique" , 
    RSS.ID_RSS AS "NUM RSS" , 
    RSS.CODE_GHM AS "CODE GHM RSS" , 
    RSS.DP_RETENU AS "DP RETENU" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSS.AGE_ENTREE AS "AGE PATIENT ANNEES ENTREE RSS" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN TB_UNITE_MEDICALE UM
         LEFT JOIN TB_UNITE_M_UNITE UMU
             LEFT JOIN TB_EG_UNITE EGU
                 LEFT JOIN TB_EG EG
                  ON  EGU.ID_EG = EG.ID_EG
              ON  UMU.ID_UNITE = EGU.ID_UNITE
          ON  UM.ID_UNITE_MED = UMU.ID_UNITE_MED
      ON  RSS.ID_UM_RETENU = UM.ID_UNITE_MED

WHERE 
1=1 
  AND RSS.DP_RETENU  IN ('C150','C151','C152','C153','C154','C155','C158','C159','C33','C340','C341','C342','C343','C348','C349','C37','C380','C381','C382','C383','C384','C388','C398','C450','C452','C4672','C771','C780','C781','C782','C783','D001','D021','D022','D023','D381','D382','D383','D384','D385') AND SUBSTR(RSS.CODE_GHM,1,1)='C' AND RSS.AGE_ENTREE >= '18' AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
