-- catalogue CORA : Analyse Financière / Valorisation LPP -DMI, MON / evolution consommation des DMI par fournisseurs
SELECT      TB_LPP.DATE_CONSOMMATION AS "DATE CONSOMMATION LPP" , 
    UMVT.LIBELLE_UNITE AS "LIBELLE UNITE MVT" , 
    TB_LPP.CODE_LPP AS "CODE LPP" , 
    LPP.LIB_LPP AS "LIBELLE LPP" , 
    LPP_F.LIB_FOURNISSEUR AS "NOM FOURNISSEUR" , 
    sum(nvl(TB_LPP.QTE_LPP,0)) AS "QUANTITE CONSOMMEE CUMULEE" , 
    TB_LPP.PRIX_ACHAT AS "PRIX ACHAT LPP" , 
    TB_LPP.PRIX_VENTE AS "PRIX VENTE" , 
    UGLPP.nom_util || ' ' || UGLPP.prenom_util AS "SIGNATAIRE LPP" , 
    SLPP.LIBELLE_SITE AS "LIBELLE SITE CONSOMMATION LPP" 

FROM     TB_MVT M
     INNER JOIN TB_LPP TB_LPP
         INNER JOIN LPP LPP
          ON  (TB_LPP.CODE_LPP=LPP.CODE_LPP OR TB_LPP.ID_LPP_NOMENCLATURE = LPP.ID_LPP)
         LEFT JOIN LPP_FOURNISSEUR LPP_F
          ON  TB_LPP.ID_FOURNISSEUR=LPP_F.ID_FOURNISSEUR
         INNER JOIN TB_SITE SLPP
          ON  TB_LPP.ID_SITE=SLPP.ID_SITE
         LEFT JOIN Util UGLPP
          ON  TB_LPP.ID_SIGN=UGLPP.id_util
      ON  M.ID_MVT=TB_LPP.ID_MVT
     INNER JOIN TB_UNITE UMVT
      ON  M.ID_UNITE_PMSI=UMVT.ID_UNITE

WHERE 
1=1 
 AND TB_LPP.DATE_CONSOMMATION >= '01/01/2026 00:00' AND TB_LPP.DATE_CONSOMMATION <= '31/12/2026 23:59'

 GROUP BY TB_LPP.DATE_CONSOMMATION, UMVT.LIBELLE_UNITE, TB_LPP.CODE_LPP, LPP.LIB_LPP, LPP_F.LIB_FOURNISSEUR, TB_LPP.PRIX_ACHAT, TB_LPP.PRIX_VENTE, UGLPP.nom_util || ' ' || UGLPP.prenom_util, SLPP.LIBELLE_SITE
