-- catalogue CORA : Analyse Financière / Valorisation LPP -DMI, MON / analyse MON valorisées
SELECT      M.ID_UNITE_M AS "CODE UNITE MVT" , 
    UMVT.LIBELLE_UNITE AS "LIBELLE UNITE MVT" , 
    MEDC.UCD AS "CODE UCD" , 
    MEDC.LIB_UCD AS "LIBELLE UCD" , 
    UPMO.nom_util || ' ' || UPMO.prenom_util AS "PRESCRIPTEUR M-O" , 
    VueFourMO.LIB_FOURNISSEUR AS "NOM FOURNISSEUR" , 
    MO.QUANTITE AS "QUANTITE MOL ONEREUSES" , 
    MO.PRIX_ACHAT AS "MONTANT M-O" , 
    MO.NUM_LOT AS "NUM LOT M-O" 

FROM     TB_MVT M
     INNER JOIN TB_MED_PATIENT MO
         INNER JOIN MED_COUTEUX MEDC
          ON  MO.ID_MED = MEDC.ID_MED AND (MEDC.FLAG_MON = 'O' OR MEDC.FLAG_ATU = 'O' OR MEDC.FLAG_ATH = 'O' OR MEDC.FLAG_POST_ATU = 'O')
         LEFT JOIN (Select M.ID_MED, F.LIB_FOURNISSEUR From MED_COUTEUX M, LPP_FOURNISSEUR F where F.ID_FOURNISSEUR=M.ID_LABO) VueFourMO
          ON  MO.ID_MED=VueFourMO.ID_MED
         LEFT JOIN Util UPMO
          ON  MO.ID_PRESC=UPMO.id_util
      ON  M.ID_MVT=MO.ID_MVT
     INNER JOIN TB_UNITE UMVT
      ON  M.ID_UNITE_PMSI=UMVT.ID_UNITE

WHERE 
1=1 
 AND MO.DATE_DEBUT >= '01/01/2026 00:00' AND MO.DATE_DEBUT <= '31/12/2026 23:59'
