-- catalogue CORA : Analyse Financière / Valorisation activité service / valorisation des RSS par CMD - RACINE - GHM (supplements cumulés)
SELECT      RSS.CODE_CMD || RSS.CODE_GHM AS "GHM RSS" , 
    RSS.CODE_CMD AS "CODE CMD RSS" , 
    SUBSTR(RSS.CODE_GHM,-1) AS "SUFFIXE CODE GHM RSS" , 
    PLGT.LIBELLE_GHS AS "LIBELLE_GHM" , 
    PLGT.CODE_GHM_RACINE AS "RACINE GHM" , 
    UMRSS.CODE_UNITE_MED AS "CODE UNITE RETENUE RSS" , 
    UMRSS.LIBELLE_UNITE_MED AS "LIB UNITE RETENUE RSS" , 
    PRUM.CODE_POLE AS "CODE POLE RUM" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RUM" , 
    RSS.DP_RETENU AS "DP RETENU" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSS.ID_RSS AS "NB DE RSS" , 
    RSS.DUREE_SEJOUR AS "SOMME DUREES RSS" , 
    RSS.MONTANT_GHS AS "MNT GHS" , 
    RSS.NB_JOUR_EXB AS "NB JOURS EXB" , 
    RSS.MONTANT_REDU AS "MNT EXB" , 
    RSS.QUANTITE_J_DEP AS "NB JOURS EXH" , 
    RSS.MONTANT_DEP AS "MNT EXH" , 
    RSS.NB_TOTAL_SUPPLEMENT AS "NB TOTALDEP" , 
    RSS.MNT_TOTAL_SUPPLEMENT AS "MNT TOTALDEP" , 
    RSS.MNT_TOTAL AS "MNT TOTAL RSS" , 
    LDA.LIB_PMSI_REGROUP AS "Libellé DA" , 
    LGA.LIB_PMSI_REGROUP AS "Libellé GA" , 
    LGP.LIB_PMSI_REGROUP AS "Libellé GP" , 
    RSS.NB_DIALYSE AS "NB DIALYSE" , 
    RSS.MT_DIALYSE AS "MNT DIALYSE" , 
    RSS.QUANTITE_J_REA AS "NB JOURS REA" , 
    RSS.MONTANT_REA AS "MNT REA" , 
    RSS.NB_JOUR_NEO1 AS "NB J NEO 1" , 
    RSS.MNT_JOUR_NEO1 AS "MNT NEO 1" , 
    RSS.NB_JOUR_NEO2 AS "NB J NEO 2" , 
    RSS.MNT_JOUR_NEO2 AS "MNT NEO 2" , 
    RSS.NB_JOUR_NEO3 AS "NB J NEO 3" , 
    RSS.MNT_JOUR_NEO3 AS "MNT NEO 3" , 
    RSS.NB_REP AS "NB J REA PED" , 
    RSS.MNT_REP AS "MNT REA PED" , 
    RSS.MNT_PRELEV_ORGANE AS "MNT PO" , 
    RSS.NB_JOUR_SOINS_INT AS "NB J SI" , 
    RSS.MNT_JOUR_SOINS_INT AS "MNT SI" , 
    RSS.NB_JOUR_SOINS_INT_REA AS "NB J SI REA" , 
    RSS.MNT_JOUR_SOINS_INT_REA AS "MNT SI REA" , 
    RSS.NB_JOUR_SURV_CONT AS "NB J SRC" , 
    RSS.MNT_JOUR_SURV_CONT AS "MNT SRC" , 
    RSS.NB_JOUR_SRA AS "NB J SRA" , 
    RSS.MNT_SRA AS "MNT SRA" , 
    RSS.NB_APHERESE AS "NB APHERESES" , 
    RSS.MNT_APHERESE AS "MNT APHERESE" , 
    RSS.NB_CAISSON_HB AS "NB CAISSON HB" , 
    RSS.MNT_CAISSON_HB AS "MNT CAISSON HB" , 
    RSS.NB_RDTH AS "NB RDTH" , 
    RSS.MNT_RDTH AS "MNT RDTH" , 
    RSS.NB_RDTH_PED AS "NB RDTH PED" , 
    RSS.MNT_RDTH_PED AS "MNT RDTH PED" , 
    RSS.NB_DIP AS "NB DIP" , 
    RSS.MNT_DIP AS "MNT DIP" 

FROM     TB_SYNTH_RSS RSS
     LEFT JOIN (SELECT PUM2.ID_UNITE_MED, PUM2.DATE_DEBUT, PUM2.DATE_FIN, PRUMX.ID_POLE, PRUMX.CODE_POLE, PRUMX.LIBELLE_POLE, PRUMX.ACTIF     FROM     TB_UNITE_POLE_UM PUM2, TB_UNITE_POLE PRUMX     WHERE     PUM2.ID_POLE=PRUMX.ID_POLE) PRUM
      ON  ( PRUM.DATE_DEBUT IS NULL OR PRUM.DATE_DEBUT<=RSS.DATE_FIN )    AND ( PRUM.DATE_FIN IS NULL OR PRUM.DATE_FIN>=RSS.DATE_FIN ) AND  RSS.ID_UM_RETENU=PRUM.ID_UNITE_MED
     LEFT JOIN TB_UNITE_MEDICALE UMRSS
      ON  UMRSS.ID_UNITE_MED = RSS.ID_UM_RETENU
     INNER JOIN PMSI_LIB_GHS_TARIF PLGT
      ON  RSS.CODE_CMD=PLGT.CMD AND RSS.CODE_GHS=PLGT.CODE_GHS AND RSS.CODE_GHM=PLGT.CODE_GHM AND RSS.DATE_FIN >= PLGT.DATE_DEBUT AND RSS.DATE_FIN<=PLGT.DATE_FIN
     LEFT JOIN PMSI_LIB_REGROUP LDA
      ON  PLGT.CODE_PMSI_REGROUP_DA=LDA.CODE_PMSI_REGROUP AND LDA.TYPE_PMSI_REGROUP='DA'
     LEFT JOIN PMSI_LIB_REGROUP LGA
      ON  PLGT.CODE_PMSI_REGROUP_GA=LGA.CODE_PMSI_REGROUP AND LGA.TYPE_PMSI_REGROUP='GA'
     LEFT JOIN PMSI_LIB_REGROUP LGP
      ON  PLGT.CODE_PMSI_REGROUP_GP=LGP.CODE_PMSI_REGROUP AND LGP.TYPE_PMSI_REGROUP='GP'

WHERE 
1=1 
  AND RSS.DATE_FIN >= '01/01/2026 00:00' AND RSS.DATE_FIN <= '31/12/2026 23:59'
