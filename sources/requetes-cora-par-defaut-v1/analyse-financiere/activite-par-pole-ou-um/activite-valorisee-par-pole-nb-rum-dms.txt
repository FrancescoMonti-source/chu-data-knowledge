-- catalogue CORA : Analyse Financière / Activité par Pole ou UM / activité valorisée par pole nb RUM - DMS
SELECT      PRUM.CODE_POLE AS "CODE POLE RUM" , 
    PRUM.LIBELLE_POLE AS "LIBELLE POLE RUM" , 
    RUM.MNT_TMJ AS "MONTANT TMJ RUM" , 
    RUM.MONTANT_GHM AS "MONTANT GHM RUM" , 
    RUM.DUREE AS "DUREE" , 
    RUM.ID_RUM AS "Nb RUM" 

FROM     TB_SYNTH_RUM RUM,
    TB_UNITE_POLE_UM PUM,
    TB_UNITE_POLE PRUM

WHERE 
1=1 
  AND RUM.ID_UNITE_MEDICALE=PUM.ID_UNITE_MED AND PUM.ID_POLE=PRUM.ID_POLE AND (PUM.DATE_DEBUT IS NULL OR PUM.DATE_DEBUT<=RUM.DATE_FIN_RUM) AND (PUM.DATE_FIN IS NULL OR PUM.DATE_FIN>=RUM.DATE_FIN_RUM) AND RUM.DATE_FIN_RUM>='01/01/2026 00:00' AND RUM.DATE_FIN_RUM<='31/12/2026 23:59'
