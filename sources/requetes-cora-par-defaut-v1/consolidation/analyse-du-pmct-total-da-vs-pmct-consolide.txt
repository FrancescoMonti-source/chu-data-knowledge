-- catalogue CORA : Consolidation / analyse du PMCT total DA vs PMCT consolidé
SELECT      RSS.ID_RSS AS "ID_RSS" , 
    PLGT.CODE_PMSI_REGROUP_DA AS "CODE DA" , 
    VueRssDa.LIB_PMSI_REGROUP AS "LIB DA" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    SUM(nvl(RSS.MONTANT_GHS,0)+nvl(RSS.MONTANT_DEP,0)-nvl(RSS.MONTANT_COEF,0)+nvl(RSS.MONTANT_URG,0)+nvl(RSS.MONTANT_REA,0)-nvl(RSS.MONTANT_REDU,0)+nvl(RSS.MT_24Z05Z,0)+nvl(RSS.MT_24Z06Z,0)+nvl(RSS.MT_24Z07Z,0)+nvl(RSS.MT_DIALYSE,0)+nvl(RSS.MNT_JOUR_NEO1,0)+nvl(RSS.MNT_JOUR_NEO2,0)+nvl(RSS.MNT_JOUR_NEO3,0)+nvl(RSS.MNT_JOUR_SOINS_INT,0)+nvl(RSS.MNT_JOUR_SURV_CONT,0)+nvl(RSS.MNT_JOUR_SOINS_PALL,0)+nvl(RSS.MNT_JOUR_SOINS_INT_REA,0)+nvl(RSS.MNT_PRELEV_ORGANE,0)+nvl(RSS.MNT_SRA,0)+nvl(RSS.MNT_REP,0)+nvl(RSS.MNT_CAISSON_HB,0)+nvl(RSS.MNT_APHERESE,0)+nvl(RSS.MNT_DIP,0)+nvl(RSS.MNT_RDTH,0)+nvl(RSS.MNT_ANTEPARTUM,0)+nvl(RSS.MNT_RDTH_PED,0)) AS "SOMME MNT TOTAL RSS" , 
    count(distinct RSS.ID_RSS) AS "NB DE RSS" , 
    RSS.DUREE_SEJOUR AS "DUREE CORRIGEE RSS" , 
    CNSLID_DA.PMCT AS "PMCT DA RUM" , 
    CNSLID_DA.TMJ AS "TMJ DA RUM" , 
    CNSLID_DA.DMS AS "DMS DA RUM" , 
    SUM(RSS.MONTANT_GHS) AS "SOMME MNT GHS RSS" , 
    RSS.MNT_TMJ AS "MONTANT TMJ RSS" , 
    RSS.MNT_TMJ_GHS AS "MONTANT TMJ GHS" 

FROM     TB_SEJOUR S
     INNER JOIN TB_SYNTH_RSS RSS
         LEFT JOIN PMSI_LIB_GHS_TARIF PLGT
          ON  RSS.CODE_CMD=PLGT.CMD AND RSS.CODE_GHS=PLGT.CODE_GHS AND RSS.CODE_GHM=PLGT.CODE_GHM AND RSS.DATE_FIN >= PLGT.DATE_DEBUT and RSS.DATE_FIN<=PLGT.DATE_FIN
         LEFT JOIN (SELECT L.LIB_PMSI_REGROUP, T.CMD, T.CODE_GHS, T.CODE_GHM, T.DATE_DEBUT, T.DATE_FIN FROM PMSI_LIB_GHS_TARIF T LEFT  JOIN  PMSI_LIB_REGROUP L ON L.CODE_PMSI_REGROUP = T.CODE_PMSI_REGROUP_DA AND L.TYPE_PMSI_REGROUP = 'DA') VueRssDa
          ON  RSS.CODE_CMD=VueRssDa.CMD AND RSS.CODE_GHS=VueRssDa.CODE_GHS AND RSS.CODE_GHM=VueRssDa.CODE_GHM AND RSS.DATE_FIN>=VueRssDa.DATE_DEBUT and RSS.DATE_FIN<=VueRssDa.DATE_FIN
         LEFT JOIN (SELECT DISTINCT PMCT, PMCT_OBJECTIF, TMJ, TMJ_OBJECTIF, DMS, DMS_OBJECTIF, CMD, CODE_GHS, CODE_GHM, ID_EJ FROM TB_CONSOLIDATION CNSLID, PMSI_LIB_GHS_TARIF TARIF WHERE CNSLID.TYPE_CONSO='DA' AND TARIF.CODE_PMSI_REGROUP_DA=CNSLID.VALEUR_CONSO) CNSLID_DA
          ON  RSS.CODE_GHS=CNSLID_DA.CODE_GHS  AND RSS.CODE_GHM=CNSLID_DA.CODE_GHM  AND RSS.CODE_CMD=CNSLID_DA.CMD
      ON  S.ID_SEJOUR=RSS.ID_SEJOUR

WHERE 
1=1 
AND (S.ID_EJ = CNSLID_DA.ID_EJ OR CNSLID_DA.ID_EJ IS NULL)
 AND (RSS.DATE_FIN >= '01/01/2015 00:00' AND RSS.DATE_FIN <= '07/07/2015 23:59')

 GROUP BY RSS.ID_RSS, PLGT.CODE_PMSI_REGROUP_DA, VueRssDa.LIB_PMSI_REGROUP, RSS.DATE_FIN, RSS.DUREE_SEJOUR, CNSLID_DA.PMCT, CNSLID_DA.TMJ, CNSLID_DA.DMS, RSS.MNT_TMJ, RSS.MNT_TMJ_GHS
