-- catalogue CORA : Consolidation / analyse du PMCT total CMD vs PMCT consolidé
SELECT      RSS.ID_RSS AS "ID_RSS" , 
    RSS.CODE_CMD AS "CODE CMD RSS" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSS.MONTANT_GHS AS "MONTANT GHS RSS" , 
    nvl(RSS.MONTANT_GHS,0)+nvl(RSS.MONTANT_DEP,0)-nvl(RSS.MONTANT_COEF,0)+nvl(RSS.MONTANT_URG,0)+nvl(RSS.MONTANT_REA,0)-nvl(RSS.MONTANT_REDU,0)+nvl(RSS.MT_24Z05Z,0)+nvl(RSS.MT_24Z06Z,0)+nvl(RSS.MT_24Z07Z,0)+nvl(RSS.MT_DIALYSE,0)+nvl(RSS.MNT_JOUR_NEO1,0)+nvl(RSS.MNT_JOUR_NEO2,0)+nvl(RSS.MNT_JOUR_NEO3,0)+nvl(RSS.MNT_JOUR_SOINS_INT,0)+nvl(RSS.MNT_JOUR_SURV_CONT,0)+nvl(RSS.MNT_JOUR_SOINS_PALL,0)+nvl(RSS.MNT_JOUR_SOINS_INT_REA,0)+nvl(RSS.MNT_PRELEV_ORGANE,0)+nvl(RSS.MNT_SRA,0)+nvl(RSS.MNT_REP,0)+nvl(RSS.MNT_CAISSON_HB,0)+nvl(RSS.MNT_APHERESE,0)+nvl(RSS.MNT_DIP,0)+nvl(RSS.MNT_RDTH,0)+nvl(RSS.MNT_ANTEPARTUM,0)+nvl(RSS.MNT_RDTH_PED,0) AS "MONTANT TOTAL RSS" , 
    count(distinct RSS.ID_RSS) AS "NB DE RSS" , 
    CNSLID_CMD.PMCT AS "PMCT CMD" , 
    RSS.DUREE_SEJOUR AS "DUREE CORRIGEE RSS" , 
    CNSLID_CMD.TMJ AS "TMJ CMD" , 
    CNSLID_CMD.DMS AS "DMS CMD" , 
    RSS.MNT_TMJ AS "MONTANT TMJ RSS" , 
    RSS.MNT_TMJ_GHS AS "MONTANT TMJ GHS" 

FROM     TB_SEJOUR S
     INNER JOIN TB_SYNTH_RSS RSS
         LEFT JOIN TB_CONSOLIDATION CNSLID_CMD
          ON  CNSLID_CMD.TYPE_CONSO='CMD' AND RSS.CODE_CMD=CNSLID_CMD.VALEUR_CONSO
      ON  S.ID_SEJOUR=RSS.ID_SEJOUR

WHERE 
1=1 
AND (S.ID_EJ = CNSLID_CMD.ID_EJ OR CNSLID_CMD.ID_EJ IS NULL)
 AND (RSS.DATE_FIN >= '01/01/2015 00:00' AND RSS.DATE_FIN <= '31/12/2015 23:59')

 GROUP BY RSS.ID_RSS, RSS.CODE_CMD, RSS.DATE_FIN, RSS.MONTANT_GHS, nvl(RSS.MONTANT_GHS,0)+nvl(RSS.MONTANT_DEP,0)-nvl(RSS.MONTANT_COEF,0)+nvl(RSS.MONTANT_URG,0)+nvl(RSS.MONTANT_REA,0)-nvl(RSS.MONTANT_REDU,0)+nvl(RSS.MT_24Z05Z,0)+nvl(RSS.MT_24Z06Z,0)+nvl(RSS.MT_24Z07Z,0)+nvl(RSS.MT_DIALYSE,0)+nvl(RSS.MNT_JOUR_NEO1,0)+nvl(RSS.MNT_JOUR_NEO2,0)+nvl(RSS.MNT_JOUR_NEO3,0)+nvl(RSS.MNT_JOUR_SOINS_INT,0)+nvl(RSS.MNT_JOUR_SURV_CONT,0)+nvl(RSS.MNT_JOUR_SOINS_PALL,0)+nvl(RSS.MNT_JOUR_SOINS_INT_REA,0)+nvl(RSS.MNT_PRELEV_ORGANE,0)+nvl(RSS.MNT_SRA,0)+nvl(RSS.MNT_REP,0)+nvl(RSS.MNT_CAISSON_HB,0)+nvl(RSS.MNT_APHERESE,0)+nvl(RSS.MNT_DIP,0)+nvl(RSS.MNT_RDTH,0)+nvl(RSS.MNT_ANTEPARTUM,0)+nvl(RSS.MNT_RDTH_PED,0), CNSLID_CMD.PMCT, RSS.DUREE_SEJOUR, CNSLID_CMD.TMJ, CNSLID_CMD.DMS, RSS.MNT_TMJ, RSS.MNT_TMJ_GHS
