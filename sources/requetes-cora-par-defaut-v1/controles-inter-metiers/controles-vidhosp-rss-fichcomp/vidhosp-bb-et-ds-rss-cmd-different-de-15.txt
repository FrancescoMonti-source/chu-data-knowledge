-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / VIDHOSP bb et ds RSS CMD!=15
SELECT      V.ID_SEJOUR AS "ID_SEJOUR" , 
    V.NO_SECU AS "NO_SECU" , 
    V.NO_SEJOUR_MERE AS "NO_SEJOUR_MERE" , 
    RSS.NO_SEJOUR AS "n° Séjour" , 
    RSS.CODE_CMD AS "CODE_CMD" 

FROM     TB_VIDHOSP V,
    TB_SYNTH_RSS RSS

WHERE 
1=1 
  AND V.NO_SEJOUR_MERE IS NOT NULL AND RSS.ID_SEJOUR = V.ID_SEJOUR AND RSS.CODE_CMD !='15' AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
