-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / valorisation AME
SELECT      RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    VH.FACTURABLE_AM AS "RSS - VH FACT AM" , 
    VH.MOTIF_NON_FACT AS "RSS - VH MOTIF NON FACT" , 
    VH.MNT_FACT_FORF_JOUR AS "RSS - VH MNT FACT FORF J" , 
    VH.MNT_TOT_REMB_AMC AS "RSS - VH MNT TOT REMB" 

FROM     TB_SYNTH_RSS RSS,
    TB_VIDHOSP VH

WHERE 
1=1 
  AND VH.ID_SEJOUR = RSS.ID_SEJOUR AND VH.FACTURABLE_AM = '0' AND VH.MOTIF_NON_FACT = '1' AND RSS.DATE_FIN>='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
