-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / VIDHOSP PIEB et pas dans RSS
SELECT      V.ID_SEJOUR AS "ID_SEJOUR" , 
    V.NO_SECU AS "NO_SECU" , 
    RSS.NO_SEJOUR AS "NO_SEJOUR" , 
    V.FACTURABLE_AM AS "FACTURABLE_AM" , 
    RSS.MODE_ENTREE AS "MODE_ENTREE" , 
    RSS.MODE_SORTIE AS "MODE_SORTIE" 

FROM     TB_VIDHOSP V,
    TB_SYNTH_RSS RSS

WHERE 
1=1 
  AND RSS.ID_SEJOUR = V.ID_SEJOUR AND V.FACTURABLE_AM ='0' AND V.MOTIF_NON_FACT='5' AND (RSS.MODE_ENTREE!='0' OR RSS.MODE_SORTIE!='0') AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
