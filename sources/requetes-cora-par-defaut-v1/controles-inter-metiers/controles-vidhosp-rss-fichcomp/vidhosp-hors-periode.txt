-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / VIDHOSP hors periode
SELECT      RSS.ID_RSS AS "NUM RSS" , 
    RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    RSS.DATE_DEBUT AS "DATE DEBUT RSS" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSS.MODE_ENTREE AS "MODE ENTREE RSS" , 
    RSS.CODE_PROVENANCE AS "CODE PROV RSS" , 
    VH.DATE_HOSPIT AS "RSS - VH DATE HOSPIT" , 
    VH.FACTURABLE_AM AS "RSS - VH FACT AM" 

FROM     TB_SYNTH_RSS RSS,
    TB_VIDHOSP VH

WHERE 
1=1 
  AND RSS.ID_SEJOUR=VH.ID_SEJOUR AND VH.DATE_HOSPIT >RSS.DATE_DEBUT AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
