-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / VIDHOSP admis pour PO et de RSS pas Dcd
SELECT      V.ID_SEJOUR AS "ID_SEJOUR" , 
    V.NO_SECU AS "NO_SEJOUR" , 
    RSS.NO_SEJOUR AS "N° Séjour" , 
    V.FACTURABLE_AM AS "AMO" , 
    RSS.MODE_ENTREE AS "ME" , 
    RSS.MODE_SORTIE AS "MS" , 
    RSS.CODE_GHS AS "GHS" , 
    RSS.CODE_CMD AS "CMD" , 
    V.HOSPIT_PRELEV_ORGA AS "PO" 

FROM     TB_VIDHOSP V,
    TB_SYNTH_RSS RSS

WHERE 
1=1 
  AND RSS.ID_SEJOUR=V.ID_SEJOUR AND V.HOSPIT_PRELEV_ORGA='1' AND RSS.MODE_SORTIE!='9' AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
