-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / valorisation IVG
SELECT      RSS.NO_SEJOUR AS "N° SEJOUR" , 
    RSS.DUREE_SEJOUR AS "DUREE RSS" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSS.CODE_CMD || RSS.CODE_GHM AS "GHM" , 
    CASE WHEN VH.FACTURABLE_AM='1' THEN 'AMO' WHEN VH.FACTURABLE_AM='0' THEN 'NON AMO' WHEN VH.FACTURABLE_AM='2'      THEN 'ATTENTE PEC' WHEN VH.FACTURABLE_AM='3' THEN 'ATTENTE DROITS' END AS "AMO" , 
    VH.TAUX_REMB AS "RSS - TX AMO" , 
    max( DA.ID_ACTIVITE) AS "CODE ACTIVITE" , 
    CASE WHEN MAX(A2.CODE_ACTE)='JNJP001' THEN 'AMF' ELSE (CASE WHEN MAX(DA.ID_ACTIVITE)='4' THEN (CASE WHEN RSS.DUREE_SEJOUR =0 THEN 'AMD'              WHEN RSS.DUREE_SEJOUR =1 THEN 'APD'              WHEN RSS.DUREE_SEJOUR =2 THEN 'APD' END)     ELSE (CASE WHEN RSS.DUREE_SEJOUR=0 THEN 'IMD'               WHEN RSS.DUREE_SEJOUR =1 THEN 'IPD'                WHEN RSS.DUREE_SEJOUR =2 THEN 'IPD' END ) END ) END AS "PRESTATION" , 
    CASE WHEN MAX(A2.CODE_ACTE)='JNJP001' THEN 257.91 ELSE (CASE WHEN MAX(DA.ID_ACTIVITE)='4' THEN      (CASE WHEN RSS.DUREE_SEJOUR =0 THEN 586.01 WHEN RSS.DUREE_SEJOUR =1 THEN 644.71 WHEN RSS.DUREE_SEJOUR =2 THEN 644.71 END)      ELSE (CASE WHEN RSS.DUREE_SEJOUR=0 THEN 437.3 WHEN RSS.DUREE_SEJOUR =1 THEN 477.66 WHEN RSS.DUREE_SEJOUR =2 THEN 477.66 END)END)END AS "TARIF" 

FROM     TB_SYNTH_RSS RSS,
    TB_SYNTH_RUM RUM,
    TB_MVT_RUM MVTRUM,
    TB_MVT M
     LEFT JOIN TB_ACTE A
      ON  M.ID_MVT=A.ID_MVT
     LEFT JOIN TB_ACTE A2
      ON  M.ID_MVT=A2.ID_MVT AND A2.CODE_ACTE='JNJP001',
    TB_DET_ACTE DA,
    TB_VIDHOSP VH

WHERE 
1=1 
  AND RSS.ID_RSS=RUM.ID_RSS AND RSS.CODE_CMD || RSS.CODE_GHM = '14Z08Z' AND RUM.ID_RUM=MVTRUM.ID_RUM AND MVTRUM.ID_MVT=M.ID_MVT AND A.ID_ACTE=DA.ID_ACTE AND DA.ETAT IS NULL AND RSS.ID_SEJOUR=VH.ID_SEJOUR AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'

 GROUP BY RSS.NO_SEJOUR, RSS.DUREE_SEJOUR, RSS.DATE_FIN, RSS.CODE_CMD || RSS.CODE_GHM, CASE WHEN VH.FACTURABLE_AM='1' THEN 'AMO' WHEN VH.FACTURABLE_AM='0' THEN 'NON AMO' WHEN VH.FACTURABLE_AM='2' THEN 'ATTENTE PEC'   WHEN VH.FACTURABLE_AM='3' THEN 'ATTENTE DROITS' END, VH.TAUX_REMB
