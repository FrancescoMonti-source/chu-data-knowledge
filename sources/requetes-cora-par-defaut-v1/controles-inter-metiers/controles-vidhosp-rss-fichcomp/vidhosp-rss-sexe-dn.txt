-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / VIDHOSP RSS sexe dn
SELECT      RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    P.IPP_PATIENT AS "NUM IPP PATIENT" , 
    P.SEXE AS "SEXE" , 
    P.DATE_NAIS AS "DATE NAISSANCE" , 
    VH.SEXE AS "RSS - VH SEXE" , 
    VH.DATE_NAIS AS "RSS - VH DATE NAIS" 

FROM     TB_PATIENT P,
    TB_SEJOUR S,
    TB_SYNTH_RSS RSS,
    TB_VIDHOSP VH

WHERE 
1=1 
  AND P.ID_PATIENT=S.ID_PATIENT AND S.ID_SEJOUR=RSS.ID_SEJOUR AND RSS.ID_SEJOUR=VH.ID_SEJOUR AND (P.DATE_NAIS != VH.DATE_NAIS OR (VH.SEXE ='2' AND (P.SEXE!='F' OR (VH.SEXE ='1' AND P.SEXE!='M')))) AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
