-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / PIE B de RSS et pad de VIDHOSP
SELECT      V.ID_SEJOUR AS "ID_SEJOUR" , 
    V.NO_SECU AS "No Immat" , 
    RSS.NO_SEJOUR AS "IEP" , 
    V.FACTURABLE_AM AS "Facturable AMO" , 
    V.MOTIF_NON_FACT AS "MOTIF NON AMO" , 
    RSS.MODE_ENTREE AS "ME" , 
    RSS.MODE_SORTIE AS "MS" 

FROM     TB_VIDHOSP V,
    TB_SYNTH_RSS RSS

WHERE 
1=1 
  AND RSS.ID_SEJOUR = V.ID_SEJOUR AND RSS.MODE_ENTREE='0' AND RSS.MODE_SORTIE='0' AND (V.FACTURABLE_AM !='0' OR V.MOTIF_NON_FACT!='5') AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
