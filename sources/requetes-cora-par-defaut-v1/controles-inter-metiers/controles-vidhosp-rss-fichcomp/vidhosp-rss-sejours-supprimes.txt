-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / VIDHOSP RSS sejours supprimés
SELECT      RSS.NO_SEJOUR AS "NO_SEJOUR" , 
    V.ID_SEJOUR AS "ID_SEJOUR" , 
    V.NO_SECU AS "No IMMAT" , 
    V.FACTURABLE_AM AS "FACTURABLE AMO" , 
    RSS.MODE_ENTREE||RSS.MODE_SORTIE AS "ME_MS" , 
    RSS.CODE_GHS AS "GHS" , 
    RSS.CODE_CMD AS "CMD" , 
    V.HOSPIT_PRELEV_ORGA AS "Indic PO" 

FROM     TB_VIDHOSP V,
    TB_SYNTH_RSS RSS

WHERE 
1=1 
  AND RSS.ID_SEJOUR=V.ID_SEJOUR AND (RSS.CODE_CMD ='90' OR RSS.MODE_ENTREE||RSS.MODE_SORTIE='00' OR RSS.CODE_GHS='9999' OR V.FACTURABLE_AM='0' OR V.FACTURABLE_AM='3' OR V.HOSPIT_PRELEV_ORGA='1') AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
