-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / RSS sans VIDHOSP
SELECT      RSS.ID_RSS AS "ID_RSS" , 
    NO_SEJOUR AS "NO_SEJOUR" , 
    DATE_DEBUT AS "DEBUT" , 
    DATE_FIN AS "FIN" 

FROM     TB_SYNTH_RSS RSS

WHERE 
1=1 
  AND ID_SEJOUR NOT IN (SELECT ID_SEJOUR  FROM  TB_VIDHOSP V  WHERE  V.ID_SEJOUR=RSS.ID_SEJOUR) AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
