-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / VIDHOSP repartition code exo
SELECT      RSS.ID_RSS AS "ID_RSS" , 
    VH.NO_SEJOUR AS "RSS - VH NO SEJOUR" , 
    VH.CODE_EXO AS "RSS - VH CODE EXO" , 
    VH.FACTURABLE_AM AS "RSS - VH FACT AM" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN TB_VIDHOSP VH
      ON  RSS.ID_SEJOUR=VH.ID_SEJOUR

WHERE 
1=1 
 AND VH.FACTURABLE_AM <> '0'
