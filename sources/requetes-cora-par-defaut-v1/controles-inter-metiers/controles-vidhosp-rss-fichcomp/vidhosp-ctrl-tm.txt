-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / VIDHOSP Ctrl TM
SELECT      RSS.ID_RSS AS "ID_RSS" , 
    VH.NO_SEJOUR AS "RSS - VH NO SEJOUR" , 
    VH.MNT_FACT_TIC_MODER AS "RSS - VH MNT FACT TIC MOD" , 
    VH.TAUX_REMB AS "RSS - VH TAUX REMB" , 
    RSS.DATE_FIN AS "DATE FIN RSS" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN TB_VIDHOSP VH
      ON  RSS.ID_SEJOUR=VH.ID_SEJOUR

WHERE 
1=1 
 AND RSS.DATE_FIN >= '01/01/2015 00:00' AND RSS.DATE_FIN <= '16/07/2015 23:59' AND ((VH.MNT_FACT_TIC_MODER > 0 AND VH.TAUX_REMB = '100') OR (VH.MNT_FACT_TIC_MODER = 0 AND VH.TAUX_REMB <> '100'))
