-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / FICHCOMP PO non VALO
SELECT      RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    RSS.DUREE_SEJOUR AS "DUREE RSS" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    RSS.CODE_CMD || RSS.CODE_GHM AS "GHM" 

FROM     TB_SYNTH_RSS RSS

WHERE 
1=1 
  AND MNT_PRELEV_ORGANE>0 AND NOT EXISTS (SELECT S.ID_SEJOUR FROM TB_SEJOUR_PREORG S WHERE S.ID_SEJOUR=RSS.ID_SEJOUR) AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
