-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / RSS sortie par transfert sans étab de transfert dans VIDHOSP
SELECT      RSS.ID_RSS AS "ID_RSS" , 
    RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    RSS.MODE_SORTIE AS "MODE SORTIE RSS" , 
    VH.FINESS_ETAB_TRANSFERT AS "RSS - VH FINESS ETAB TRANSFERT" , 
    RSS.DATE_FIN AS "DATE FIN RSS" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN TB_VIDHOSP VH
      ON  RSS.ID_SEJOUR=VH.ID_SEJOUR

WHERE 
1=1 
 AND VH.FINESS_ETAB_TRANSFERT IS NULL AND RSS.DATE_FIN >= '01/01/2014 00:00' AND RSS.DATE_FIN <= '12/04/2015 23:59' AND RSS.MODE_SORTIE = '7'
