-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / VIDHOSP mère et ds RSS CMD!=14
SELECT      V.ID_SEJOUR AS "ID_SEJOUR" , 
    V.NO_SECU AS "NO_SECU" , 
    V.NO_SEJOUR_MERE AS "IEP_MERE" , 
    RSS.CODE_CMD AS "CMD" 

FROM     TB_VIDHOSP V,
    TB_SYNTH_RSS RSS

WHERE 
1=1 
  AND V.NO_SEJOUR_MERE IS NOT NULL AND RSS.NO_SEJOUR = V.NO_SEJOUR_MERE AND RSS.CODE_CMD! ='14' AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
