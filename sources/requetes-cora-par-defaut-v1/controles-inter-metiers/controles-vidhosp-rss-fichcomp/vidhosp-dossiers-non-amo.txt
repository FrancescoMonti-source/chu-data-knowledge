-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / VIDHOSP dossiers non AMO
SELECT      RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    VH.FACTURABLE_AM AS "RSS - VH FACT AM" , 
    RSS.ID_RSS AS "NUM RSS" , 
    CASE  VH.MOTIF_NON_FACT WHEN  '1' THEN 'AME' WHEN '2' THEN 'CVI' WHEN '3' THEN 'PAYANT' WHEN '4' THEN 'URG' WHEN '5' THEN 'PIE' WHEN '6' THEN 'AL' WHEN '9' THEN 'AUTRE' ELSE 'NR' END AS "RSS - VH MOTIF NON FACT" , 
    RSS.DATE_FIN AS "Sorti le" 

FROM     TB_SYNTH_RSS RSS,
    TB_VIDHOSP VH

WHERE 
1=1 
  AND rss.id_sejour = vh.id_sejour AND (VH.FACTURABLE_AM = '0' OR VH.FACTURABLE_AM ='3') AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
