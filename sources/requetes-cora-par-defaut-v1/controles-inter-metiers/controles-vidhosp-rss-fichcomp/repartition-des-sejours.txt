-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / repartition des sejours
SELECT      RSS.ID_RSS AS "NO RSS" , 
    RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    CASE WHEN VH.FACTURABLE_AM = 1 THEN 'AMO' WHEN VH.FACTURABLE_AM = 2 THEN 'ATTENTE PEC'  WHEN VH.FACTURABLE_AM =3 THEN 'ATTENTE DROITS'  ELSE 'NON AMO' END AS "RSS - VH FACT AM" , 
    RSS.ID_RSS AS "NUM RSS" , 
    CASE WHEN VH.MOTIF_NON_FACT = '1' THEN 'AME' WHEN VH.MOTIF_NON_FACT = '2' THEN 'CVI' WHEN VH.MOTIF_NON_FACT = '3' THEN 'PAYANT' WHEN VH.MOTIF_NON_FACT = '4' THEN 'URG' WHEN VH.MOTIF_NON_FACT = '5' THEN 'PIE' WHEN VH.MOTIF_NON_FACT = '6' THEN 'AL' WHEN VH.MOTIF_NON_FACT = '9' THEN 'AUTRE'  ELSE VH.MOTIF_NON_FACT  END AS "RSS - VH MOTIF NON FACT" , 
    RSS.DATE_FIN AS "Sorti le" 

FROM     TB_SYNTH_RSS RSS,
    TB_VIDHOSP VH

WHERE 
1=1 
  AND RSS.ID_SEJOUR = VH.ID_SEJOUR AND RSS.DATE_FIN>='01/01/2026 00:00' AND RSS.DATE_FIN<='31/12/2026 23:59'
