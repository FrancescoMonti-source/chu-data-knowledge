-- catalogue CORA : Controles inter metiers / Controles VIDHOSP RSS FICHCOMP / FICHCOMP RSS en dehors période
SELECT      RSS.ID_RSS AS "NUM RSS" , 
    RSS.NO_SEJOUR AS "NO SEJOUR RSS" , 
    RSS.DATE_DEBUT AS "DATE DEBUT RSS" , 
    RSS.DATE_FIN AS "DATE FIN RSS" , 
    M.ID_MVT AS "IDENTIFIANT MVT" , 
    MO.DATE_DEBUT AS "DATE ADMINISTRATION" , 
    MO.PRIX_ACHAT AS "MONTANT M-O" 

FROM     TB_SYNTH_RSS RSS,
    TB_SYNTH_RUM RUM,
    TB_MVT_RUM MVTRUM,
    TB_MVT M,
    TB_MED_PATIENT MO

WHERE 
1=1 
  AND RSS.ID_RSS=RUM.ID_RSS AND RUM.ID_RUM=MVTRUM.ID_RUM AND MVTRUM.ID_MVT=M.ID_MVT AND M.ID_MVT=MO.ID_MVT AND (MO.DATE_DEBUT > RSS.DATE_FIN OR MO.DATE_DEBUT < RSS.DATE_DEBUT) AND RSS.DATE_FIN >='01/01/2026 00:00' AND RSS.DATE_FIN <='31/12/2026 23:59'
