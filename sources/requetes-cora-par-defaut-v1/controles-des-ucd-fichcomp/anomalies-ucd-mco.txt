-- catalogue CORA : Controles des UCD FICHCOMP / anomalies UCD (MCO)
SELECT      RSS.ID_RSS AS "ID_RSS" , 
    RSS.DATE_DEBUT AS "DATE_DEBUT" , 
    RSS.DATE_FIN AS "DATE_FIN" , 
    M.ID_MVT AS "ID_MVT" , 
    U.FINESS_GEO AS "FINESS_GEO" , 
    S.NO_SEJOUR AS "NO_SEJOUR" , 
    MC.CODE_UCD13 AS "CODE UCD 13" , 
    MC.UCD AS "CODE UCD 7" , 
    PRIX_ACHAT AS "PRIX_PAYE" , 
    MP.QUANTITE AS "QTE" , 
    MP.CODE_LES AS "INDICATION" , 
    CASE  WHEN MP.TYPE_INDICATION='M' THEN 'MON' WHEN MP.TYPE_INDICATION='P' THEN 'POST ATU' WHEN MP.TYPE_INDICATION='A' THEN 'ATU' WHEN MP.TYPE_INDICATION='I' THEN 'INTRA' END AS "TYPE INDIC DOS" , 
    MP.DATE_DEBUT AS "DATE ADMINISTRATION" , 
    (SELECT VI.DATE_FIN                    FROM                   CORA_CCAM.VIEW_UCD_INDICATION VI WHERE VI.UCD=MC.UCD AND VI.CODE_LES = MP.CODE_LES) AS "FIN VALIDITE" , 
    (SELECT  CASE   WHEN VI.FLAG_INDICATION='M' THEN 'MON'  WHEN VI.FLAG_INDICATION='P' THEN 'POST ATU'  WHEN VI.FLAG_INDICATION='A' THEN 'ATU'  WHEN VI.FLAG_INDICATION='I' THEN 'INTRA'  END                      FROM                    CORA_CCAM.VIEW_UCD_INDICATION VI WHERE VI.UCD=MC.UCD AND VI.CODE_LES = MP.CODE_LES) AS "TYPE INDIC REF" , 
    CASE          WHEN MP.CODE_LES IS NULL  THEN 'UCD en sus sans indication'   WHEN NOT EXISTS (SELECT 1             FROM             CORA_CCAM.VIEW_UCD_INDICATION VI WHERE VI.UCD=MC.UCD AND VI.CODE_LES = MP.CODE_LES) THEN 'Anomalie Indication non valide : Indication n''appartient pas au référentiel'    WHEN NOT EXISTS (SELECT 1             FROM             CORA_CCAM.VIEW_UCD_INDICATION VI  WHERE VI.UCD=MC.UCD AND VI.CODE_LES = MP.CODE_LES AND (VI.DATE_DEBUT <=  MP.DATE_DEBUT Or VI.DATE_DEBUT Is NULL) And (VI.DATE_FIN >= MP.DATE_DEBUT Or  VI.DATE_FIN Is NULL)) THEN 'Anomalie Indication non valide : Code indication non valide dans le référentiel à la date de pose'     WHEN MP.CODE_LES IS NOT NULL AND MP.TYPE_INDICATION Is NULL AND EXISTS (SELECT 1                    FROM                    CORA_CCAM.VIEW_UCD_INDICATION VI WHERE VI.UCD=MC.UCD AND VI.CODE_LES = MP.CODE_LES AND VI.FLAG_INDICATION IN ('A','P')) THEN  'Anomalie Typage : Indication ATU non typée'         WHEN MP.CODE_LES IS NOT NULL AND MP.TYPE_INDICATION Is NULL AND EXISTS (SELECT 1                    FROM                    CORA_CCAM.VIEW_UCD_INDICATION VI WHERE VI.UCD=MC.UCD AND VI.CODE_LES = MP.CODE_LES AND VI.FLAG_INDICATION IN ('I')) THEN  'Anomalie Typage : Indication Intra GHS non typée'      WHEN MP.CODE_LES IS NOT NULL AND MP.TYPE_INDICATION Is NULL AND EXISTS (SELECT 1                    FROM                    CORA_CCAM.VIEW_UCD_INDICATION VI WHERE VI.UCD=MC.UCD AND VI.CODE_LES = MP.CODE_LES AND VI.FLAG_INDICATION IN ('M')) THEN  'Anomalie Typage : Indication MON non typée'   WHEN MP.CODE_LES IS NOT NULL AND MP.TYPE_INDICATION IS NOT NULL AND EXISTS (SELECT 1                    FROM                    CORA_CCAM.VIEW_UCD_INDICATION VI WHERE VI.UCD=MC.UCD AND VI.CODE_LES = MP.CODE_LES AND VI.FLAG_INDICATION<>MP.TYPE_INDICATION) THEN  'Anomalie Typage : différence de typage entre indication sur le dossier et dans le référentiel'  ELSE 'Anomalie : Autre'      END AS "TYPE ANOMALIE" 

FROM     TB_SYNTH_RSS RSS
     INNER JOIN TB_SEJOUR S
      ON  (RSS.ID_SEJOUR = S.ID_SEJOUR)
     INNER JOIN TB_MVT M
      ON  (M.ID_SEJOUR = S.ID_SEJOUR AND M.ID_RSS = RSS.ID_RSS)
     INNER JOIN TB_UNITE U
      ON  (M.ID_UNITE_PMSI = U.ID_UNITE)
     INNER JOIN TB_MED_PATIENT MP
      ON  (M.ID_MVT = MP.ID_MVT)
     INNER JOIN CORA_CCAM.MED_COUTEUX MC
      ON  (MC.ID_MED = MP.ID_MED)

WHERE 
1=1 
  AND ((MP.CODE_LES IS NULL AND ((MC.FLAG_MON = 'O' And ((MC.DATE_DEBUT_MON <=  MP.DATE_DEBUT Or MC.DATE_DEBUT_MON Is NULL)   And (MC.DATE_FIN_MON >= MP.DATE_DEBUT Or  MC.DATE_FIN_MON Is NULL))) OR ((MC.FLAG_ATU = 'O' AND (MC.DATE_DEBUT_ATU <= MP.DATE_DEBUT OR MC.DATE_DEBUT_ATU Is null) AND (MC.DATE_FIN_ATU >= MP.DATE_DEBUT OR MC.DATE_FIN_ATU Is Null)) OR (MC.FLAG_POST_ATU = 'O' AND (MC.DATE_DEBUT_POST_ATU <= MP.DATE_DEBUT Or MC.DATE_DEBUT_POST_ATU Is null) And (MC.DATE_FIN_POST_ATU >= MP.DATE_DEBUT Or MC.DATE_FIN_POST_ATU Is Null)))) AND ((NVL(U.URGENCE, 'N')='N' AND MP.DATE_DEBUT>=RSS.DATE_DEBUT AND MP.DATE_DEBUT<RSS.DATE_FIN + 1) OR URGENCE='O') ) OR ( (MP.CODE_LES IS NOT NULL AND MP.CODE_LES NOT IN ('I999998','I999999','NXXXX00')) AND NOT EXISTS (SELECT 1             FROM             CORA_CCAM.VIEW_UCD_INDICATION VI             WHERE             VI.UCD=MC.UCD AND VI.CODE_LES = MP.CODE_LES AND VI.FLAG_INDICATION=MP.TYPE_INDICATION AND (VI.DATE_DEBUT <=  MP.DATE_DEBUT Or VI.DATE_DEBUT Is NULL) And (VI.DATE_FIN >= MP.DATE_DEBUT Or  VI.DATE_FIN Is NULL) ) AND ((NVL(U.URGENCE, 'N')='N' AND MP.DATE_DEBUT>=RSS.DATE_DEBUT AND MP.DATE_DEBUT<RSS.DATE_FIN + 1) OR URGENCE='O'))) AND RSS.DATE_FIN >= '01/01/2023 00:00' AND RSS.DATE_FIN <= '31/12/2023 23:59'
